// Reading a food label from a photo: what the model is asked for, and how
// its answer is matched to the food list. The call itself is in cloud.js.
// No DOM access here, so the tests can run it.

import { ALLERGENS, CATEGORIES } from "./data/meta.js";
import { normalizeText } from "./logic.js";

export const LABEL_MODEL = "gemini-3.8-flash";

export const LABEL_PROMPT = `This photo shows the packaging of a food for a baby or young child, often an Australian product. Read only what is printed; never guess.

- readable: false if the photo doesn't show enough of a food label to read.
- name: the product as a parent would say it, e.g. "Potato & beef purée". Leave out slogans and age claims.
- brand: the brand name.
- category: "mixed" for meals, pouches and anything made of several main foods. Otherwise the one category the food belongs to.
- ingredientsText: the ingredient list exactly as printed.
- ingredients: the foods in it as plain everyday words, most first, e.g. ["potato", "beef", "carrot"]. Leave out water, salt, vitamins, minerals, thickeners, acidity regulators and other additives.
- contains: allergens the label says the food contains, from a "Contains:" statement or allergens in bold in the ingredients. Crustaceans and molluscs count as shellfish. Almonds, cashews, hazelnuts and other tree nuts count as treenut.
- mayContain: allergens in a "may contain" or "may be present" statement.
- otherAllergens: anything else in the allergen statements that doesn't fit the list, e.g. lupin.`;

// The answer's shape, built with the SDK's Schema helpers.
export function labelSchema(Schema) {
  const allergen = () => Schema.array({ items: Schema.enumString({ enum: ALLERGENS.map((a) => a.id) }) });
  return Schema.object({
    properties: {
      readable: Schema.boolean(),
      name: Schema.string(),
      brand: Schema.string(),
      category: Schema.enumString({ enum: CATEGORIES.map((c) => c.id) }),
      ingredientsText: Schema.string(),
      ingredients: Schema.array({ items: Schema.string() }),
      contains: allergen(),
      mayContain: allergen(),
      otherAllergens: Schema.array({ items: Schema.string() }),
    },
    optionalProperties: ["name", "brand", "category", "ingredientsText", "ingredients", "contains", "mayContain", "otherAllergens"],
  });
}

// Words in an ingredient list that point to one of the nine allergens: a
// second check, in case the model misses one the label shows.
const ALLERGEN_WORDS = {

export function spotAllergens(text) {
  const words = ` ${normalizeText(text || "")} `;
  return ALLERGENS.map((a) => a.id).filter((id) => ALLERGEN_WORDS[id].some((w) => words.includes(` ${w} `)));
}

// Label words that shouldn't count as a food tried.
const NOT_FOODS = new Set(["water", "salt", "sugar"]);

// Simple plurals, so "carrots" finds Carrot and "tomatoes" finds Tomato.
function variants(text) {
  const words = text.split(" ");
  return [text, words.map((w) => w.replace(/oes$/, "o").replace(/ies$/, "y").replace(/s$/, "")).join(" ")];
}

// Match the label's ingredients ("beef mince", "sweet potato") to listed
// foods. The longest food name found inside an ingredient wins, so "sweet
// potato" finds Sweet potato rather than Potato.
export function matchIngredients(names, foods) {
  const index = foods.map((f) => ({ food: f, keys: [f.name, ...(f.aka || [])].map(normalizeText).filter(Boolean) }));
  const matched = [];
  const unmatched = [];
  for (const raw of names || []) {
    const text = normalizeText(raw);
    if (!text || NOT_FOODS.has(text)) continue;
    let best = null;
    let bestLength = 0;
    for (const candidate of variants(text)) {
      for (const { food, keys } of index) {
        for (const key of keys) {
          if (key.length > bestLength && ` ${candidate} `.includes(` ${key} `)) {
            best = food;
            bestLength = key.length;
          }
        }
      }
    }
    if (!best) unmatched.push(String(raw).trim());
    else if (!matched.includes(best.id)) matched.push(best.id);
  }
  return { matched, unmatched };
}

const ALLERGEN_IDS = new Set(ALLERGENS.map((a) => a.id));
const CATEGORY_IDS = new Set(CATEGORIES.map((c) => c.id));
const strings = (value) => (Array.isArray(value) ? value.filter((v) => typeof v === "string" && v.trim()).map((v) => v.trim()) : []);
const allergenIds = (value) => [...new Set(strings(value).filter((a) => ALLERGEN_IDS.has(a)))];

// Turn the model's answer into what the add-a-food form needs.
export function parseLabel(answer, foods) {
  const a = answer && typeof answer === "object" ? answer : {};
  const text = typeof a.ingredientsText === "string" ? a.ingredientsText.trim().slice(0, 2000) : "";
  const { matched, unmatched } = matchIngredients(strings(a.ingredients).slice(0, 30), foods);
  const contains = allergenIds(a.contains);
  // Allergen words in the printed ingredients that the model didn't list.
  const spotted = spotAllergens(text).filter((id) => !contains.includes(id));
  return {
    readable: a.readable !== false,
    name: typeof a.name === "string" ? a.name.trim().slice(0, 60) : "",
    brand: typeof a.brand === "string" ? a.brand.trim().slice(0, 60) : "",
    category: CATEGORY_IDS.has(a.category) ? a.category : "",
    ingredientsText: text,
    ingredientIds: matched,
    unmatched,
    contains: [...contains, ...spotted],
    spotted,
    mayContain: allergenIds(a.mayContain).filter((id) => !contains.includes(id) && !spotted.includes(id)),
    otherAllergens: strings(a.otherAllergens).slice(0, 10),
  };
}

import { test } from "node:test";
import assert from "node:assert/strict";
import { FOODS } from "../js/data/index.js";
import { labelSchema, matchIngredients, parseLabel, spotAllergens } from "../js/label.js";

test("label ingredients are matched to listed foods", () => {
  const { matched, unmatched } = matchIngredients(["Beef mince", "sweet potato", "Carrots", "tomatoes", "water", "thickener (maize starch)", "onion powder"], FOODS);
  assert.deepEqual(matched.slice(0, 4), ["ground-beef", "sweet-potato", "carrot", "tomato"], "beef mince is ground beef");
  assert.ok(!matched.includes("potato"), "sweet potato isn't potato");
  assert.ok(!unmatched.includes("water"), "water is skipped");
});

test("allergen words in the printed ingredients are spotted", () => {
  assert.deepEqual(spotAllergens("Pumpkin (40%), wholemeal WHEAT flour, skim milk powder, whey, almonds"), ["milk", "treenut", "wheat"]);
  assert.deepEqual(spotAllergens("Apple, pear, banana"), []);
  assert.deepEqual(spotAllergens("Contains: crustacean"), ["shellfish"]);
});

test("parseLabel merges the model's answer with a second allergen check", () => {
  const read = parseLabel(
    {
      readable: true,
      name: "Potato & Beef Purée",
      brand: "Little Spoon",
      category: "mixed",
      ingredientsText: "Potato (55%), beef (25%), water, butter, onion",
      ingredients: ["potato", "beef", "water", "butter", "onion"],
      contains: ["soy", "lupin"],
      mayContain: ["egg", "soy"],
      otherAllergens: ["lupin"],
    },
    FOODS,
  );
  assert.equal(read.name, "Potato & Beef Purée");
  assert.deepEqual(read.ingredientIds, ["potato", "beef", "butter", "onion"]);
  assert.deepEqual(read.contains, ["soy", "milk"], "butter in the text adds milk; unknown ids are dropped");
  assert.deepEqual(read.spotted, ["milk"]);
  assert.deepEqual(read.mayContain, ["egg"], "already-contained allergens aren't repeated");
  assert.deepEqual(read.otherAllergens, ["lupin"]);
  assert.equal(parseLabel({ readable: false }, FOODS).readable, false);
  assert.equal(parseLabel({ category: "snacks" }, FOODS).category, "");
});

test("the label schema builds with the SDK's helpers", () => {
  const calls = [];
  const fake = new Proxy({}, { get: (_, kind) => (params) => (calls.push(kind), { kind, ...params }) });
  const schema = labelSchema(fake);
  assert.equal(schema.kind, "object");
  assert.ok(schema.properties.contains.items.enum.includes("shellfish"));
  assert.ok(schema.properties.category.enum.includes("mixed"));
});

// Small helper so each food entry stays compact in the data files.
//
// food(id, name, category, {
//   age       earliest age in months to offer it (default 6), or null to avoid
//   wait      why to wait, when age > 6 or age is null
//   aka       other names, used by search
//   allergens ids from ALLERGENS
//   choking   "low" | "prep" | "high"
//   iron      true when it is a good source of iron
//   s6/s9/s12 how to serve it at each stage, in plain words
//   tips      short safety and nutrition notes
//   ss        Solid Starts page slug (defaults to id); null for no direct link
//   icon      tile icon key (defaults to js/data/food-icons.js, then the category icon)
// })
export function food(id, name, cat, o = {}) {
  return {
    id,
    name,
    cat,
    aka: o.aka || [],
    age: o.age === undefined ? 6 : o.age,
    wait: o.wait || "",
    allergens: o.allergens || [],
    choking: o.choking || "low",
    iron: !!o.iron,
    serve: { 6: o.s6 || "", 9: o.s9 || "", 12: o.s12 || "" },
    tips: o.tips || [],
    ss: o.ss === undefined ? id : o.ss,
    icon: o.icon || "",
  };
}

// Loads Firebase only when js/sync-config.js has a config, and wraps the few
// calls the app needs: sign-in state, sign in/out, password reset.

// Kept in variables so bundlers leave them alone: the SDK files are only
// downloaded when syncing is configured, and the label reader only when
// someone reads a label.
const SDK_PATH = "./vendor/firebase.js";
const AI_SDK_PATH = "./vendor/firebase-ai.js";

// The config file is edited by hand, so load it on its own: a typo there
// should turn syncing off with a message, not stop the whole app loading.
// Resolves { config } when sync is set up, { config: null } when it is
// switched off, or { error } when the file can't be used.
export async function readSyncConfig() {
  let mod;
  try {
    mod = await import("./sync-config.js");
  } catch {
    return { config: null, error: "js/sync-config.js has a mistake in it. It should contain only the settings between { and } from Firebase." };
  }
  const config = mod.FIREBASE_CONFIG;
  if (config == null) return { config: null };
  if (typeof config !== "object" || !config.apiKey || !config.projectId) {
    return { config: null, error: "js/sync-config.js is missing the apiKey or projectId from Firebase." };
  }
  // Label reading is on once a reCAPTCHA site key is filled in.
  const reading = mod.LABEL_READING;
  const labels =
    reading && typeof reading.recaptchaSiteKey === "string" && reading.recaptchaSiteKey.trim()
      ? { recaptchaSiteKey: reading.recaptchaSiteKey.trim(), provider: reading.provider === "recaptcha-enterprise" ? "recaptcha-enterprise" : "recaptcha-v3", model: reading.model || null }
      : null;
  return { config, labels };
}

let FIREBASE_CONFIG = null;
let loading = null;

export function loadCloud(config) {
  if (config) FIREBASE_CONFIG = config;
  if (!FIREBASE_CONFIG) return Promise.reject(new Error("Syncing isn't set up."));
  loading ||= (async () => {
    const fb = await import(SDK_PATH);
    const app = fb.initializeApp(FIREBASE_CONFIG);
    const auth = fb.getAuth(app);
    let db;
    try {
      // Offline cache: tries logged without a connection sync when it returns.
      db = fb.initializeFirestore(app, { localCache: fb.persistentLocalCache({ tabManager: fb.persistentMultipleTabManager() }) });
    } catch {
      db = fb.getFirestore(app);
    }
    return {
      fb,
      db,
      app,
      onUser: (callback) => fb.onAuthStateChanged(auth, callback),
      signIn: (email, password) => fb.signInWithEmailAndPassword(auth, email, password),
      signOut: () => fb.signOut(auth),
      resetPassword: (email) => fb.sendPasswordResetEmail(auth, email),
    };
  })();
  loading.catch(() => (loading = null));
  return loading;
}

// Reading a food label from a photo, with Gemini through Firebase AI Logic.
// App Check (with a reCAPTCHA key) proves the requests come from this site.
let labelModel = null;
let appCheckOn = false;

export async function readLabelImage(cloud, settings, { prompt, schema, model }, base64) {
  labelModel ||= (async () => {
    const ai = await import(AI_SDK_PATH);
    if (!appCheckOn) {
      const Provider = settings.provider === "recaptcha-enterprise" ? ai.ReCaptchaEnterpriseProvider : ai.ReCaptchaV3Provider;
      ai.initializeAppCheck(cloud.app, { provider: new Provider(settings.recaptchaSiteKey), isTokenAutoRefreshEnabled: true });
      appCheckOn = true;
    }
    const service = ai.getAI(cloud.app, { backend: new ai.GoogleAIBackend() });
    return ai.getGenerativeModel(service, {
      model: settings.model || model,
      generationConfig: { responseMimeType: "application/json", responseSchema: schema(ai.Schema) },
    });
  })();
  labelModel.catch(() => (labelModel = null));
  const gemini = await labelModel;
  const result = await gemini.generateContent([prompt, { inlineData: { data: base64, mimeType: "image/jpeg" } }]);
  return JSON.parse(result.response.text());
}

// Messages for what can go wrong reading a label.
export function describeLabelError(err) {
  const code = String((err && err.code) || "");
  const message = String((err && err.message) || "");
  const status = err && err.customErrorData && err.customErrorData.status;
  if (code === "unreadable") return "Couldn't find a label to read in that photo. Try again closer up, in good light, or type it in.";
  if (typeof navigator !== "undefined" && navigator.onLine === false) return "You're offline. Type it in, or read the label when you're back online.";
  if (code.includes("api-not-enabled")) return "Label reading isn't switched on in Firebase yet (AI Logic). See “Reading food labels from a photo” in the README.";
  if (code.startsWith("appCheck/") || /app ?check/i.test(message) || status === 401 || status === 403) {
    return "This site isn't set up for label reading yet (App Check). See “Reading food labels from a photo” in the README.";
  }
  if (status === 429 || /quota|resource.?exhausted|rate limit/i.test(message)) return "That's all the label reads the free plan allows for now. Try again later, or type it in.";
  if (err instanceof SyntaxError) return "Couldn't make sense of that photo. Try again, or type it in.";
  return "Couldn't read the label. Try again, or type it in.";
}

export function describeAuthError(err) {
  const code = (err && err.code) || "";
  if (["auth/invalid-credential", "auth/wrong-password", "auth/user-not-found", "auth/invalid-login-credentials"].includes(code)) {
    return "That email and password don't match. Check them, or reset your password.";
  }
  if (code === "auth/invalid-email") return "Enter a valid email address.";
  if (code === "auth/missing-password") return "Enter your password.";
  if (code === "auth/too-many-requests") return "Too many attempts. Wait a few minutes and try again.";
  if (code === "auth/network-request-failed") return "No connection. Check your internet and try again.";
  if (code === "auth/user-disabled") return "This account has been turned off in Firebase.";
  return (err && err.message) || "Couldn't sign in. Try again.";
}

import { ALLERGENS, ALLERGEN_DUE_DAYS, CATEGORIES, OUTCOMES, STAGES } from "./data/meta.js";

const ALLERGEN_IDS = new Set(ALLERGENS.map((a) => a.id));
const CATEGORY_IDS = new Set(CATEGORIES.map((c) => c.id));

const DAY_MS = 24 * 60 * 60 * 1000;

export function emptyState() {
  return { logs: {}, queue: {}, profile: {}, foods: {} };

// ---------- foods the family adds ----------

// Their ids start with "my-", so they can't clash with the built-in foods.
const CUSTOM_ID = /^my-[a-z0-9-]{1,60}$/;

export function isCustomId(id) {
  return typeof id === "string" && CUSTOM_ID.test(id);
}

export function newCustomId(name) {
  const slug = normalizeText(name).replace(/ /g, "-").slice(0, 30) || "food";
  return `my-${slug}-${Math.random().toString(36).slice(2, 8)}`;
}

// Is this food id one a log or to-try entry may point at?
export function isKnownFoodId(id, knownFoodIds) {
  return typeof id === "string" && (!knownFoodIds || knownFoodIds.has(id) || isCustomId(id));
}

const text = (value, max) => (typeof value === "string" ? value.trim().slice(0, max) : "");
const allergenList = (value) => (Array.isArray(value) ? [...new Set(value.filter((a) => ALLERGEN_IDS.has(a)))] : []);

// Validate a food the family added, from storage or a backup.
export function cleanCustomFood(raw, knownFoodIds) {
  if (!raw || typeof raw !== "object" || !isCustomId(raw.id)) return null;
  const name = text(raw.name, 60);
  if (!name || !CATEGORY_IDS.has(raw.cat)) return null;
  const ingredients = Array.isArray(raw.ingredients) ? raw.ingredients.filter((id) => typeof id === "string" && !isCustomId(id) && (!knownFoodIds || knownFoodIds.has(id))) : [];
  const food = {
    id: raw.id,
    name,
    cat: raw.cat,
    ingredients: [...new Set(ingredients)].slice(0, 30),
    allergens: allergenList(raw.allergens),
    mayContain: allergenList(raw.mayContain),
    brand: text(raw.brand, 60),
    notes: text(raw.notes, 1000),
    label: text(raw.label, 2000),
    createdAt: Number.isFinite(raw.createdAt) ? raw.createdAt : 0,
    updatedAt: Number.isFinite(raw.updatedAt) ? raw.updatedAt : 0,
  };
  if (typeof raw.by === "string" && raw.by) food.by = raw.by.slice(0, 200);
  if (typeof raw.byName === "string" && raw.byName) food.byName = raw.byName.slice(0, 60);
  return food;
}

const CHOKING_RANK = { low: 0, prep: 1, high: 2 };

// Turn a family food into the same shape as a built-in one. Age, choking
// risk, iron and allergens come from what's in it, so a pouch with honey
// says "From 12 months" and one with egg counts as egg.
export function expandCustomFood(custom, foodsById, categoryIcons) {
  const parts = custom.ingredients.map((id) => foodsById.get(id)).filter(Boolean);
  const avoid = parts.filter((f) => f.age == null);
  const oldest = parts.filter((f) => f.age != null).sort((a, b) => b.age - a.age);
  const age = avoid.length ? null : Math.max(6, ...parts.map((f) => f.age));
  let wait = "";
  if (avoid.length) wait = `Contains ${avoid.map((f) => f.name.toLowerCase()).join(" and ")}, which is best avoided for young children.`;
  else if (age > 6) {
    const late = oldest.filter((f) => f.age === age).map((f) => f.name.toLowerCase());
    wait = `Contains ${late.join(" and ")}, which isn't recommended until ${formatAgeRule({ age }).replace("From ", "")}.`;
  }
  const choking = parts.length ? parts.reduce((worst, f) => (CHOKING_RANK[f.choking] > CHOKING_RANK[worst] ? f.choking : worst), "low") : null;
  const allergens = ALLERGENS.map((a) => a.id).filter((a) => custom.allergens.includes(a) || parts.some((f) => f.allergens.includes(a)));
  return {
    id: custom.id,
    name: custom.name,
    cat: custom.cat,
    aka: custom.brand ? [custom.brand] : [],
    age,
    wait,
    allergens,
    choking,
    iron: parts.some((f) => f.iron),
    serve: { 6: "", 9: "", 12: "" },
    tips: [],
    ss: null,
    icon: parts.length === 1 ? parts[0].icon : categoryIcons[custom.cat],
    custom: true,
    ingredients: custom.ingredients,
    mayContain: custom.mayContain.filter((a) => !allergens.includes(a)),
    brand: custom.brand,
    notes: custom.notes,
    label: custom.label,
    source: custom,
  };
}

// Foods tried toward the goal: a family food made of listed foods counts as
// those foods, not as one more.
export function countsAsFood(food) {
  return !!food && !(food.custom && food.ingredients.length);
}

export function todayISO(now = new Date()) {
  return toISODate(now);
}

// The date n days before today, in the phone's own time zone.
export function daysAgoISO(n, now = new Date()) {
  return toISODate(new Date(now.getFullYear(), now.getMonth(), now.getDate() - n));
}

export function toISODate(d) {
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${y}-${m}-${day}`;
}

function parseISODate(iso) {
  if (typeof iso !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(iso)) return null;
  const [y, m, d] = iso.split("-").map(Number);
  const date = new Date(y, m - 1, d);
  return date.getMonth() === m - 1 ? date : null;
}

export function daysBetween(fromISO, toISO) {
  const a = parseISODate(fromISO);
  const b = parseISODate(toISO);
  if (!a || !b) return null;
  return Math.round((b - a) / DAY_MS);
}

// Whole months between birth date and `today`, or null if unknown.
export function ageInMonths(birthISO, todayIso = todayISO()) {
  const birth = parseISODate(birthISO);
  const now = parseISODate(todayIso);
  if (!birth || !now || now < birth) return null;
  let months = (now.getFullYear() - birth.getFullYear()) * 12 + (now.getMonth() - birth.getMonth());
  if (now.getDate() < birth.getDate()) months -= 1;
  return Math.max(0, months);
}

export function formatAge(months) {
  if (months == null) return "";
  if (months < 24) return `${months} mo`;
  const years = Math.floor(months / 12);
  const rest = months % 12;
  return rest ? `${years} yr ${rest} mo` : `${years} yr`;
}

// The same age for use in a sentence: "4 months", "1 year 3 months".
export function formatAgeLong(months) {
  if (months == null) return "";
  const unit = (n, word) => `${n} ${word}${n === 1 ? "" : "s"}`;
  if (months < 24) return unit(months, "month");
  const rest = months % 12;
  return rest ? `${unit(Math.floor(months / 12), "year")} ${unit(rest, "month")}` : unit(months / 12, "year");
}

// The serving stage to show by default for a baby of this age.
export function stageForAge(months) {
  if (months == null) return STAGES[0].id;
  let stage = STAGES[0];
  for (const s of STAGES) if (months >= s.from) stage = s;
  return stage.id;
}

export function formatAgeRule(food) {
  if (food.age == null) return "Avoid";
  if (food.age >= 24 && food.age % 12 === 0) return `From ${food.age / 12} years`;
  return `From ${food.age} months`;
}

// Can a baby of this age have this food yet? Babies under 6 months see the
// foods they can start with.
export function isReadyAt(food, months) {
  if (food.age == null) return false;
  return food.age <= Math.max(months ?? 6, 6);
}

const OUTCOME_TONE = Object.fromEntries(OUTCOMES.map((o) => [o.id, o.tone]));

function compareLogs(a, b) {
  if (a.date !== b.date) return a.date < b.date ? -1 : 1;
  return (a.createdAt || 0) - (b.createdAt || 0);
}

export function logsForFood(state, foodId) {
  return Object.values(state.logs)
    .filter((l) => l.foodId === foodId)
    .sort(compareLogs);
}

// status: "new" | "liked" | "learning" | "reaction". The latest try decides
// it, so a food offered again without symptoms moves on from "reaction"; the
// earlier reaction stays in reactionCount and lastReactionDate. Tries that
// came from a meal (viaMeal) count too, but a reaction to a meal stays with
// the meal: here it's only listed in mealReactions.
export function summarizeLogs(logs) {
  if (!logs.length) return { count: 0, status: "new" };
  const sorted = [...logs].sort(compareLogs);
  const last = sorted[sorted.length - 1];
  const reactions = sorted.filter((l) => (l.reactions || []).length > 0);
  const lastTone = OUTCOME_TONE[last.outcome];
  let status = lastTone === "good" ? "liked" : "learning";
  if ((last.reactions || []).length) status = "reaction";
  return {
    count: sorted.length,
    status,
    firstDate: sorted[0].date,
    lastDate: last.date,
    lastOutcome: last.outcome,
    reactionCount: reactions.length,
    lastReactionDate: reactions.length ? reactions[reactions.length - 1].date : null,
    mealCount: sorted.filter((l) => l.viaMeal).length,
    mealReactions: sorted.filter((l) => l.mealReaction).map((l) => ({ mealId: l.viaMeal, date: l.date })),
    lovedCount: sorted.filter((l) => l.outcome === "loved").length,
  };
}

export function summarizeAll(state) {
  const foods = state.foods || {};
  const grouped = new Map();
  const add = (id, log) => {
    if (!grouped.has(id)) grouped.set(id, []);
    grouped.get(id).push(log);
  };
  for (const log of Object.values(state.logs)) {
    if (!isCustomId(log.foodId)) {
      add(log.foodId, log);
      continue;
    }
    const food = foods[log.foodId];
    if (!food) continue; // tries of a food that has since been deleted
    add(log.foodId, log);
    // A meal is also a try of each thing in it.
    for (const id of food.ingredients) add(id, { ...log, reactions: [], viaMeal: log.foodId, mealReaction: (log.reactions || []).length > 0 });
  }
  const out = new Map();
  for (const [foodId, logs] of grouped) out.set(foodId, summarizeLogs(logs));
  return out;
}

export function allergenProgress(foods, summaries, todayIso = todayISO()) {
  return ALLERGENS.map((a) => {
    const related = foods.filter((f) => f.allergens.includes(a.id));
    const tried = related.filter((f) => summaries.has(f.id));
    const lastDate = tried.reduce((max, f) => {
      const d = summaries.get(f.id).lastDate;
      return !max || d > max ? d : max;
    }, null);
    const reaction = tried.some((f) => summaries.get(f.id).status === "reaction");
    const daysSince = lastDate ? daysBetween(lastDate, todayIso) : null;
    let state = "new";
    if (reaction) state = "reaction";
    else if (lastDate) state = daysSince > ALLERGEN_DUE_DAYS ? "due" : "ok";
    return { ...a, foods: related, tried, lastDate, daysSince, state };
  });
}

// Up to `limit` ideas for what to offer next: allergens that are due again,
// then one new allergen (from 6 months), the to-try list, and iron-rich foods.
export function suggestNext({ foods, summaries, queue, months, today = todayISO(), firstFoods, ironStarters, limit = 4 }) {
  const byId = new Map(foods.map((f) => [f.id, f]));
  const out = [];
  const add = (id, kind, extra = {}) => {
    if (out.length < limit && byId.has(id) && !out.some((s) => s.id === id)) out.push({ id, kind, ...extra });
  };
  const allergens = allergenProgress(foods, summaries, today);
  for (const a of allergens.filter((x) => x.state === "due").sort((x, y) => y.daysSince - x.daysSince)) {
    const latest = [...a.tried].sort((x, y) => (summaries.get(x.id).lastDate < summaries.get(y.id).lastDate ? 1 : -1))[0];
    add(latest.id, "due", { allergen: a.id, daysSince: a.daysSince });
  }
  if (months == null || months >= 6) {
    const fresh = allergens.find((a) => a.state === "new");
    if (fresh) add(firstFoods[fresh.id], "allergen", { allergen: fresh.id });
  }
  for (const q of Object.values(queue).sort((a, b) => a.addedAt - b.addedAt)) add(q.foodId, "queued");
  for (const id of ironStarters) {
    const food = byId.get(id);
    if (food && !summaries.has(id) && isReadyAt(food, months)) add(id, "iron");
  }
  return out;
}

export function categoryProgress(foods, summaries) {
  return CATEGORIES.map((c) => {
    const inCat = foods.filter((f) => f.cat === c.id);
    return { ...c, total: inCat.length, tried: inCat.filter((f) => summaries.has(f.id)).length };
  });
}

// Count of distinct foods first tried before the baby's first birthday.
export function triedBeforeFirstBirthday(summaries, birthISO) {
  const birth = parseISODate(birthISO);
  if (!birth) return null;
  const firstBirthday = toISODate(new Date(birth.getFullYear() + 1, birth.getMonth(), birth.getDate()));
  let n = 0;
  for (const s of summaries.values()) if (s.firstDate < firstBirthday) n += 1;
  return n;
}

export function normalizeText(s) {
  return String(s)
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
}

export function matchesQuery(food, query) {
  const q = normalizeText(query);
  if (!q) return true;
  const haystack = normalizeText([food.name, ...food.aka].join(" "));
  return q.split(" ").every((word) => haystack.includes(word));
}

export function newId() {
  if (globalThis.crypto && typeof crypto.randomUUID === "function") return crypto.randomUUID();
  return `id-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`;
}

const OUTCOME_IDS = new Set(OUTCOMES.map((o) => o.id));
const ISO_DATE = /^\d{4}-\d{2}-\d{2}$/;

// Validate and clean a log entry from an untrusted source (import file,
// shared store). Returns null when it can't be used.
export function cleanLog(raw, knownFoodIds) {
  if (!raw || typeof raw !== "object") return null;
  const { id, foodId, date, outcome } = raw;
  if (typeof id !== "string" || !id || id.length > 100) return null;
  if (!isKnownFoodId(foodId, knownFoodIds)) return null;
  if (typeof date !== "string" || !ISO_DATE.test(date)) return null;
  if (!OUTCOME_IDS.has(outcome)) return null;
  const log = {
    id,
    foodId,
    date,
    outcome,
    reactions: Array.isArray(raw.reactions) ? raw.reactions.filter((r) => typeof r === "string").slice(0, 10) : [],
    served: typeof raw.served === "string" ? raw.served.slice(0, 300) : "",
    notes: typeof raw.notes === "string" ? raw.notes.slice(0, 2000) : "",
    createdAt: Number.isFinite(raw.createdAt) ? raw.createdAt : 0,
    updatedAt: Number.isFinite(raw.updatedAt) ? raw.updatedAt : 0,
  };
  if (typeof raw.by === "string" && raw.by) log.by = raw.by.slice(0, 200);
  if (typeof raw.byName === "string" && raw.byName) log.byName = raw.byName.slice(0, 60);
  return log;
}

export function cleanProfile(raw) {
  const p = {};
  if (!raw || typeof raw !== "object") return p;
  if (typeof raw.babyName === "string") p.babyName = raw.babyName.slice(0, 60);
  if (typeof raw.birthDate === "string" && (raw.birthDate === "" || ISO_DATE.test(raw.birthDate))) p.birthDate = raw.birthDate;
  if (Number.isInteger(raw.goal) && raw.goal > 0 && raw.goal <= 1000) p.goal = raw.goal;
  if (Array.isArray(raw.readySigns)) p.readySigns = raw.readySigns.filter((x) => typeof x === "string" && x.length <= 40).slice(0, 20);
  return p;
}

// Merge an exported backup into the current state. Entries are matched by
// id; when both sides have the same entry, the more recently edited wins.
// Returns { state, added, updated }.
export function mergeBackup(current, backup, knownFoodIds) {
  if (!backup || typeof backup !== "object" || backup.app !== "first-bites") {
    throw new Error("This file isn't a First Bites backup.");
  }
  const logs = { ...current.logs };
  let added = 0;
  let updated = 0;
  for (const raw of Object.values(backup.logs || {})) {
    const log = cleanLog(raw, knownFoodIds);
    if (!log) continue;
    const existing = logs[log.id];
    if (!existing) {
      logs[log.id] = log;
      added += 1;
    } else if ((log.updatedAt || 0) > (existing.updatedAt || 0)) {
      logs[log.id] = log;
      updated += 1;
    }
  }
  const queue = { ...current.queue };
  for (const [foodId, entry] of Object.entries(backup.queue || {})) {
    if (!isKnownFoodId(foodId, knownFoodIds)) continue;
    if (!queue[foodId]) queue[foodId] = { foodId, addedAt: Number(entry && entry.addedAt) || 0 };
  }
  const foods = { ...(current.foods || {}) };
  for (const raw of Object.values(backup.foods || {})) {
    const food = cleanCustomFood(raw, knownFoodIds);
    if (food && (!foods[food.id] || (food.updatedAt || 0) > (foods[food.id].updatedAt || 0))) foods[food.id] = food;
  }
  const incoming = cleanProfile(backup.profile);
  const profile = { ...current.profile };
  for (const key of Object.keys(incoming)) {
    if (profile[key] === undefined || profile[key] === "") profile[key] = incoming[key];
  }
  return { state: { logs, queue, profile, foods }, added, updated };
}

export function makeBackup(state, now = new Date()) {
  return {
    app: "first-bites",
    version: 1,
    exportedAt: now.toISOString(),
    profile: state.profile,
    logs: state.logs,
    queue: state.queue,
    foods: state.foods || {},
  };
}

// Where tracking data lives.
//
// LocalStore keeps everything in this browser's localStorage. Partners on
// different phones share by exporting a backup and importing it (merge).
//
// SharedStore is used when the page runs as a claude.ai Artifact with the
// `db` capability, and CloudStore when the app is hosted elsewhere with a
// Firebase project configured (js/sync-config.js). In both, every log entry
// is its own document, so two people logging at once never overwrite each
// other.

import { cleanCustomFood, cleanLog, cleanProfile, emptyState, isKnownFoodId, mergeBackup } from "./logic.js";

const STATE_KEY = "first-bites:v1";
const DEVICE_KEY = "first-bites:device-name";

function readJSON(key) {
  try {
    const raw = localStorage.getItem(key);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
}

function writeJSON(key, value) {
  try {
    localStorage.setItem(key, JSON.stringify(value));
    return true;
  } catch {
    return false;
  }
}

export function getDeviceName() {
  try {
    return localStorage.getItem(DEVICE_KEY) || "";
  } catch {
    return "";
  }
}

export function setDeviceName(name) {
  try {
    localStorage.setItem(DEVICE_KEY, name);
  } catch {
    // Storage unavailable (private mode); the name just won't persist.
  }
}

class Emitter {
  constructor() {
    this.listeners = new Set();
  }
  subscribe(fn) {
    this.listeners.add(fn);
    fn(this.state);
    return () => this.listeners.delete(fn);
  }
  emit() {
    for (const fn of this.listeners) fn(this.state);
  }
}

function loadLocalState(knownFoodIds) {
  const saved = readJSON(STATE_KEY);
  const state = emptyState();
  if (!saved) return state;
  for (const raw of Object.values(saved.logs || {})) {
    const log = cleanLog(raw, knownFoodIds);
    if (log) state.logs[log.id] = log;
  }
  for (const [foodId, entry] of Object.entries(saved.queue || {})) {
    if (isKnownFoodId(foodId, knownFoodIds)) state.queue[foodId] = { foodId, addedAt: Number(entry && entry.addedAt) || 0 };
  }
  for (const raw of Object.values(saved.foods || {})) {
    const food = cleanCustomFood(raw, knownFoodIds);
    if (food) state.foods[food.id] = food;
  }
  state.profile = cleanProfile(saved.profile);
  return state;
}

// This device's saved log, read without opening a store.
export function readLocalState(knownFoodIds) {
  return loadLocalState(knownFoodIds);
}

// Drop tries from this device's own log, once they've moved to the shared log.
export function removeLocalLogs(ids, knownFoodIds) {
  const state = loadLocalState(knownFoodIds);
  for (const id of ids) delete state.logs[id];
  return writeJSON(STATE_KEY, state);
}

export class LocalStore extends Emitter {
  constructor(knownFoodIds) {
    super();
    this.mode = "local";
    this.canWrite = true;
    this.knownFoodIds = knownFoodIds;
    this.state = loadLocalState(knownFoodIds);
    this.persisted = writeJSON(STATE_KEY, this.state);
    // Keep several open tabs in step.
    this.onStorage = (e) => {
      if (e.key !== STATE_KEY) return;
      this.state = loadLocalState(knownFoodIds);
      this.emit();
    };
    window.addEventListener("storage", this.onStorage);
  }

  dispose() {
    window.removeEventListener("storage", this.onStorage);
  }

  commit(next) {
    this.state = next;
    this.persisted = writeJSON(STATE_KEY, next);
    this.emit();
  }

  async me() {
    return { id: null, name: getDeviceName() };
  }

  async profiles() {
    return {};
  }

  async saveLog(log) {
    const entry = { ...log, byName: log.byName ?? getDeviceName() };
    if (!entry.byName) delete entry.byName;
    this.commit({ ...this.state, logs: { ...this.state.logs, [log.id]: entry } });
  }

  async deleteLog(id) {
    const logs = { ...this.state.logs };
    delete logs[id];
    this.commit({ ...this.state, logs });
  }

  async setQueued(foodId, on) {
    const queue = { ...this.state.queue };
    if (on) queue[foodId] = { foodId, addedAt: Date.now() };
    else delete queue[foodId];
    this.commit({ ...this.state, queue });
  }

  async setProfile(patch) {
    this.commit({ ...this.state, profile: { ...this.state.profile, ...patch } });
  }

  async saveFood(food) {
    const entry = { ...food, byName: food.byName ?? getDeviceName() };
    if (!entry.byName) delete entry.byName;
    this.commit({ ...this.state, foods: { ...this.state.foods, [food.id]: entry } });
  }

  async deleteFood(id) {
    const { logIds } = linkedTo(this.state, id);
    const foods = { ...this.state.foods };
    const logs = { ...this.state.logs };
    const queue = { ...this.state.queue };
    delete foods[id];
    delete queue[id];
    for (const logId of logIds) delete logs[logId];
    this.commit({ ...this.state, foods, logs, queue });
  }

  async importBackup(backup) {
    const { state, added, updated } = mergeBackup(this.state, backup, this.knownFoodIds);
    this.commit(state);
    return { added, updated };
  }

  async clearAll() {
    this.commit(emptyState());
  }
}

// Messages for errors a viewer can do something about.
export function describeStoreError(err) {
  const code = err && err.code;
  if (code === "permission-denied") return "This account isn't allowed to use the shared log. Check the email addresses in your Firestore rules.";
  if (code === "unauthenticated") return "You've been signed out. Sign in again on the Baby tab to keep syncing.";
  if (code === "invalid_argument") return "You can view this tracker but not change it. Ask the owner to give you edit access.";
  if (code === "quota_exceeded") return "The shared tracker is full. Export a backup and delete old entries to make room.";
  if (code === "resource_exhausted") return "Too many changes at once. Wait a moment and try again.";
  if (code === "revoked" || code === "not_granted") return "Access to the shared tracker has ended for this page.";
  return (err && err.message) || "Couldn't save that change. Check your connection and try again.";
}

async function withRetry(fn) {
  try {
    return await fn();
  } catch (err) {
    if (err && err.code === "unavailable") {
      await new Promise((r) => setTimeout(r, 400 + Math.random() * 600));
      return fn();
    }
    throw err;
  }
}

export class SharedStore extends Emitter {
  constructor(db, user, knownFoodIds) {
    super();
    this.mode = "shared";
    this.db = db;
    this.user = user;
    this.knownFoodIds = knownFoodIds;
    this.state = emptyState();
    this.loaded = { logs: false, queue: false, profile: false, foods: false };
    this.error = null;
    this.canWrite = true;
    this.onError = () => {};
    this.profileWrite = Promise.resolve();
  }

  get ready() {
    return Object.values(this.loaded).every(Boolean);
  }

  async start() {
    if (this.user) {
      const can = await this.user.can("data.write");
      if (can === false) this.canWrite = false;
      this.viewer = await this.user.me();
    }
    // A listener stops for good when it fails. Keep the error so the page
    // can say so, rather than show a log that has stopped updating.
    const fail = (err) => {
      this.error = err;
      this.emit();
    };
    this.db.collection("logs").onSnapshot((snap) => {
      const logs = {};
      for (const d of snap.docs) {
        const log = cleanLog({ ...d.data(), id: d.id }, this.knownFoodIds);
        if (log) logs[log.id] = log;
      }
      this.loaded.logs = true;
      this.state = { ...this.state, logs };
      this.emit();
    }, fail);
    this.db.collection("queue").onSnapshot((snap) => {
      const queue = {};
      for (const d of snap.docs) {
        if (isKnownFoodId(d.id, this.knownFoodIds)) queue[d.id] = { foodId: d.id, addedAt: Number(d.data().addedAt) || 0 };
      }
      this.loaded.queue = true;
      this.state = { ...this.state, queue };
      this.emit();
    }, fail);
    this.db.doc("meta/profile").onSnapshot((snap) => {
      this.loaded.profile = true;
      this.state = { ...this.state, profile: snap.exists ? cleanProfile(snap.data()) : {} };
      this.emit();
    }, fail);
    this.db.collection("foods").onSnapshot((snap) => {
      const foods = {};
      for (const d of snap.docs) {
        const food = cleanCustomFood({ ...d.data(), id: d.id }, this.knownFoodIds);
        if (food) foods[food.id] = food;
      }
      this.loaded.foods = true;
      this.state = { ...this.state, foods };
      this.emit();
    }, fail);
  }

  async me() {
    return this.viewer ? { id: this.viewer.id, name: this.viewer.name } : { id: null, name: "" };
  }

  // Resolve viewer ids stored on log entries to display names.
  async profiles(ids) {
    if (!this.user || !ids.length) return {};
    return this.user.profiles(ids);
  }

  async saveLog(log) {
    const entry = { ...log };
    if (!entry.by && this.viewer && this.viewer.id) entry.by = this.viewer.id;
    await withRetry(() => this.db.collection("logs").doc(log.id).set(entry));
  }

  async deleteLog(id) {
    await withRetry(() => this.db.collection("logs").doc(id).delete());
  }

  async setQueued(foodId, on) {
    const ref = this.db.collection("queue").doc(foodId);
    await withRetry(() => (on ? ref.set({ foodId, addedAt: Date.now() }) : ref.delete()));
  }

  // Profile edits go to one document, so queue them one at a time.
  async setProfile(patch) {
    const run = async () => {
      const next = { ...this.state.profile, ...patch };
      await withRetry(() => this.db.doc("meta/profile").set(next));
    };
    this.profileWrite = this.profileWrite.then(run, run);
    return this.profileWrite;
  }

  async saveFood(food) {
    const entry = { ...food };
    if (!entry.by && this.viewer && this.viewer.id) entry.by = this.viewer.id;
    await withRetry(() => this.db.collection("foods").doc(food.id).set(entry));
  }

  async deleteFood(id) {
    const { logIds, queued } = linkedTo(this.state, id);
    for (const logId of logIds) await this.deleteLog(logId);
    if (queued) await this.setQueued(id, false);
    await withRetry(() => this.db.collection("foods").doc(id).delete());
  }

  async importBackup(backup) {
    const { state, added, updated } = mergeBackup(this.state, backup, this.knownFoodIds);
    for (const food of Object.values(state.foods)) {
      if (this.state.foods[food.id] !== food) await withRetry(() => this.db.collection("foods").doc(food.id).set(food));
    }
    const changed = Object.values(state.logs).filter((l) => this.state.logs[l.id] !== l);
    for (const log of changed) {
      await withRetry(() => this.db.collection("logs").doc(log.id).set(log));
    }
    for (const foodId of Object.keys(state.queue)) {
      if (!this.state.queue[foodId]) await this.setQueued(foodId, true);
    }
    const profilePatch = {};
    for (const [k, v] of Object.entries(state.profile)) if (this.state.profile[k] !== v) profilePatch[k] = v;
    if (Object.keys(profilePatch).length) await this.setProfile(profilePatch);
    return { added, updated };
  }

  async clearAll() {
    for (const id of Object.keys(this.state.logs)) await this.deleteLog(id);
    for (const foodId of Object.keys(this.state.queue)) await this.setQueued(foodId, false);
    for (const id of Object.keys(this.state.foods)) await withRetry(() => this.db.collection("foods").doc(id).delete());
    await withRetry(() => this.db.doc("meta/profile").delete());
  }
}

// The tries and to-try entry that go with a family food, so deleting the
// food takes them too.
function linkedTo(state, foodId) {
  return {
    logIds: Object.values(state.logs).filter((l) => l.foodId === foodId).map((l) => l.id),
    queued: !!state.queue[foodId],
  };
}

// Firestore-backed store for the GitHub Pages version. Writes go to the local
// offline cache first, so they show up immediately and sync when online;
// we don't wait for the server before returning.
export class CloudStore extends Emitter {
  constructor(cloud, user, knownFoodIds) {
    super();
    this.mode = "cloud";
    this.canWrite = true;
    this.cloud = cloud;
    this.user = user;
    this.knownFoodIds = knownFoodIds;
    this.state = emptyState();
    this.loaded = { logs: false, queue: false, profile: false, foods: false };
    this.error = null;
    this.pendingWrites = false;
    this.onError = () => {};
    this.unsubs = [];
  }

  get ready() {
    return Object.values(this.loaded).every(Boolean);
  }

  get email() {
    return this.user.email || "";
  }

  start() {
    const { fb, db } = this.cloud;
    // A listener stops for good when it fails (e.g. the rules turn this
    // account away). Keep the error so the page can say so and offer a retry.
    const fail = (err) => {
      this.error = err;
      this.emit();
    };
    this.unsubs.push(
      fb.onSnapshot(fb.collection(db, "logs"), { includeMetadataChanges: true }, (snap) => {
        this.pendingWrites = snap.metadata.hasPendingWrites;
        // When a write reaches the server only the metadata changes: pass on
        // the same state, so the page updates the sync note but not the list.
        if (this.loaded.logs && !snap.docChanges().length) {
          this.emit();
          return;
        }
        const logs = {};
        for (const d of snap.docs) {
          const log = cleanLog({ ...d.data(), id: d.id }, this.knownFoodIds);
          if (log) logs[log.id] = log;
        }
        this.loaded.logs = true;
        this.state = { ...this.state, logs };
        this.emit();
      }, fail),
      fb.onSnapshot(fb.collection(db, "queue"), (snap) => {
        const queue = {};
        for (const d of snap.docs) {
          if (isKnownFoodId(d.id, this.knownFoodIds)) queue[d.id] = { foodId: d.id, addedAt: Number(d.data().addedAt) || 0 };
        }
        this.loaded.queue = true;
        this.state = { ...this.state, queue };
        this.emit();
      }, fail),
      fb.onSnapshot(fb.doc(db, "meta", "profile"), (snap) => {
        this.loaded.profile = true;
        this.state = { ...this.state, profile: snap.exists() ? cleanProfile(snap.data()) : {} };
        this.emit();
      }, fail),
      fb.onSnapshot(fb.collection(db, "foods"), (snap) => {
        const foods = {};
        for (const d of snap.docs) {
          const food = cleanCustomFood({ ...d.data(), id: d.id }, this.knownFoodIds);
          if (food) foods[food.id] = food;
        }
        this.loaded.foods = true;
        this.state = { ...this.state, foods };
        this.emit();
      }, fail),
    );
  }

  dispose() {
    for (const unsub of this.unsubs) unsub();
    this.unsubs = [];
  }

  restart() {
    this.dispose();
    this.error = null;
    this.loaded = { logs: false, queue: false, profile: false, foods: false };
    this.start();
  }

  // Fire-and-forget: the offline cache applies the write right away.
  send(promise) {
    promise.catch((err) => this.onError(err));
  }

  displayName() {
    return getDeviceName() || this.email.split("@")[0];
  }

  async me() {
    return { id: this.user.uid, name: this.displayName() };
  }

  async profiles() {
    return {};
  }

  async saveLog(log) {
    const { fb, db } = this.cloud;
    const entry = JSON.parse(JSON.stringify({ ...log, by: log.by || this.user.uid, byName: log.byName || this.displayName() }));
    this.send(fb.setDoc(fb.doc(db, "logs", log.id), entry));
  }

  async deleteLog(id) {
    const { fb, db } = this.cloud;
    this.send(fb.deleteDoc(fb.doc(db, "logs", id)));
  }

  async setQueued(foodId, on) {
    const { fb, db } = this.cloud;
    const ref = fb.doc(db, "queue", foodId);
    this.send(on ? fb.setDoc(ref, { foodId, addedAt: Date.now() }) : fb.deleteDoc(ref));
  }

  async setProfile(patch) {
    const { fb, db } = this.cloud;
    this.send(fb.setDoc(fb.doc(db, "meta", "profile"), patch, { merge: true }));
  }

  async saveFood(food) {
    const { fb, db } = this.cloud;
    const entry = JSON.parse(JSON.stringify({ ...food, by: food.by || this.user.uid, byName: food.byName || this.displayName() }));
    this.send(fb.setDoc(fb.doc(db, "foods", food.id), entry));
  }

  // The food, its tries and its to-try entry go together, in one batch.
  async deleteFood(id) {
    const { fb, db } = this.cloud;
    const { logIds, queued } = linkedTo(this.state, id);
    this.writeAll([
      ...logIds.map((logId) => (b) => b.delete(fb.doc(db, "logs", logId))),
      ...(queued ? [(b) => b.delete(fb.doc(db, "queue", id))] : []),
      (b) => b.delete(fb.doc(db, "foods", id)),
    ]);
  }

  // Firestore batches hold up to 500 writes.
  writeAll(ops) {
    const { fb, db } = this.cloud;
    for (let i = 0; i < ops.length; i += 450) {
      const batch = fb.writeBatch(db);
      for (const op of ops.slice(i, i + 450)) op(batch);
      this.send(batch.commit());
    }
  }

  async importBackup(backup) {
    const { fb, db } = this.cloud;
    const { state, added, updated } = mergeBackup(this.state, backup, this.knownFoodIds);
    const ops = [];
    for (const food of Object.values(state.foods)) {
      if (this.state.foods[food.id] !== food) ops.push((b) => b.set(fb.doc(db, "foods", food.id), JSON.parse(JSON.stringify(food))));
    }
    for (const log of Object.values(state.logs)) {
      if (this.state.logs[log.id] !== log) ops.push((b) => b.set(fb.doc(db, "logs", log.id), JSON.parse(JSON.stringify(log))));
    }
    for (const foodId of Object.keys(state.queue)) {
      if (!this.state.queue[foodId]) ops.push((b) => b.set(fb.doc(db, "queue", foodId), { foodId, addedAt: state.queue[foodId].addedAt || Date.now() }));
    }
    const profilePatch = {};
    for (const [k, v] of Object.entries(state.profile)) if (this.state.profile[k] !== v) profilePatch[k] = v;
    if (Object.keys(profilePatch).length) ops.push((b) => b.set(fb.doc(db, "meta", "profile"), profilePatch, { merge: true }));
    this.writeAll(ops);
    return { added, updated };
  }

  async clearAll() {
    const { fb, db } = this.cloud;
    const ops = [
      ...Object.keys(this.state.logs).map((id) => (b) => b.delete(fb.doc(db, "logs", id))),
      ...Object.keys(this.state.queue).map((id) => (b) => b.delete(fb.doc(db, "queue", id))),
      ...Object.keys(this.state.foods).map((id) => (b) => b.delete(fb.doc(db, "foods", id))),
      (b) => b.delete(fb.doc(db, "meta", "profile")),
    ];
    this.writeAll(ops);
  }
}

// Pick the store for this page. Inside a claude.ai Artifact viewer with the
// db capability granted, the tracker is shared; anywhere else it is local.
export async function openStore(knownFoodIds) {
  const claude = typeof window !== "undefined" ? window.claude : undefined;
  if (claude && typeof claude.use === "function") {
    try {
      const [db, user] = await Promise.all([claude.use("db"), claude.use("user")]);
      if (db) {
        const store = new SharedStore(db, user, knownFoodIds);
        await store.start();
        return store;
      }
    } catch {
      // Fall through to local storage.
    }
  }
  return new LocalStore(knownFoodIds);
}

import { test } from "node:test";
import assert from "node:assert/strict";
import { FOODS } from "../js/data/index.js";
import { ALLERGENS, ALLERGEN_FIRST_FOOD, ALLERGEN_STARTERS, CATEGORIES, CHOKING, IRON_STARTERS, OUTCOMES } from "../js/data/meta.js";
import { GUIDE } from "../js/data/guide.js";
import { ICONS } from "../js/data/icons.js";
import { FOOD_ICONS } from "../js/data/food-icons.js";

const categoryIds = new Set(CATEGORIES.map((c) => c.id));
const allergenIds = new Set(ALLERGENS.map((a) => a.id));

test("food ids are unique and URL-safe", () => {
  const seen = new Set();
  for (const f of FOODS) {
    assert.match(f.id, /^[a-z0-9-]+$/, f.id);
    assert.ok(!seen.has(f.id), `duplicate id ${f.id}`);
    seen.add(f.id);
  }
});

test("every food has a valid category, choking level and allergens", () => {
  for (const f of FOODS) {
    assert.ok(categoryIds.has(f.cat), `${f.id}: category ${f.cat}`);
    assert.ok(f.choking in CHOKING, `${f.id}: choking ${f.choking}`);
    for (const a of f.allergens) assert.ok(allergenIds.has(a), `${f.id}: allergen ${a}`);
    assert.ok(ICONS[f.icon], `${f.id}: icon "${f.icon}" exists`);
  }
});

test("serving text exists for every stage a food is allowed at", () => {
  for (const f of FOODS) {
    if (f.age == null) {
      assert.ok(f.wait, `${f.id}: avoided foods explain why`);
      continue;
    }
    if (f.age > 6) assert.ok(f.wait, `${f.id}: foods that wait explain why`);
    for (const [stage, from] of [["6", 6], ["9", 9], ["12", 12]]) {
      const next = stage === "6" ? 9 : stage === "9" ? 12 : Infinity;
      if (f.age < next && f.age <= 12) assert.ok(f.serve[stage], `${f.id}: serve text for ${stage}+ (from ${from})`);
    }
  }
});

test("safety essentials are encoded", () => {
  const byId = Object.fromEntries(FOODS.map((f) => [f.id, f]));
  assert.equal(byId.honey.age, 12);
  assert.equal(byId["cows-milk"].age, 12);
  assert.equal(byId.juice.age, 12);
  assert.equal(byId.popcorn.age, 48);
  assert.equal(byId.swordfish.age, null);
  for (const id of ["grape", "blueberry", "apple", "carrot", "peanut", "hot-dog", "popcorn"]) {
    assert.equal(byId[id].choking, "high", `${id} is a common choking hazard`);
  }
  assert.deepEqual(byId.eggs.allergens, ["egg"]);
  assert.deepEqual(byId.peanut.allergens, ["peanut"]);
});

test("every allergen has at least one food that can be offered at 6 months", () => {
  for (const a of ALLERGENS) {
    assert.ok(FOODS.some((f) => f.allergens.includes(a.id) && f.age === 6), a.id);
  }
});

test("suggested first foods carry their allergen, and iron starters are iron-rich", () => {
  const byId = new Map(FOODS.map((f) => [f.id, f]));
  for (const a of ALLERGENS) {
    const food = byId.get(ALLERGEN_FIRST_FOOD[a.id]);
    assert.ok(food && food.allergens.includes(a.id) && food.age === 6, a.id);
  }
  for (const id of IRON_STARTERS) assert.ok(byId.get(id)?.iron && byId.get(id).age === 6, id);
  for (const c of CATEGORIES) assert.ok(c.short, c.id);
});

test("every outcome has a face icon", () => {
  for (const o of OUTCOMES) assert.ok(ICONS[o.icon], o.id);
});

test("every allergen has first-food ideas", () => {
  for (const a of ALLERGENS) assert.ok(ALLERGEN_STARTERS[a.id], a.id);
  assert.deepEqual(Object.keys(ALLERGEN_STARTERS).sort(), ALLERGENS.map((a) => a.id).sort());
});

test("every food has its own icon mapping, and no mapping is stale", () => {
  const ids = new Set(FOODS.map((f) => f.id));
  for (const f of FOODS) assert.ok(FOOD_ICONS[f.id], `${f.id} has an entry in food-icons.js`);
  for (const id of Object.keys(FOOD_ICONS)) assert.ok(ids.has(id), `food-icons.js entry ${id} matches a food`);
});

test("Solid Starts links are slugs, not URLs", () => {
  for (const f of FOODS) if (f.ss) assert.match(f.ss, /^[a-z0-9-]+$/, f.id);
});

# First Bites

A starting-solids reference and tracker for two parents. Look up how to serve
a food at your baby's age, then log each try: when, how it went, how it was
served, and any reaction.

## What it does

- **366 foods** in 11 categories, each with an icon and:
  - the earliest age to offer it, or why to wait or avoid it
  - how to serve it at 6–8 months, 9–11 months and 12 months +
  - common allergens, choking risk, and whether it's iron-rich
  - safety and nutrition tips
  - a link to the food's page on [Solid Starts](https://solidstarts.com/foods/) for their full guide, photos and videos (356 foods have one)
- **Tracker.** Tap the circle on any food tile and pick a face (loved it → wouldn't try) to log today's try in one tap, with **Undo** if you tap the wrong one. **Add details** covers the day (Today, Yesterday or another day), how it was served (with quick picks like *mashed* or *pre-loaded spoon*), any reaction signs and notes. Each tile shows its status (liked, still learning or reaction, each with its own symbol) and how many times it's been tried.
- **Your own foods.** Add anything that isn't listed, by tapping **Add a food** (or **Add "…"** when a search finds nothing). File it under any category, e.g. kangaroo under Meat & poultry, or under the new **Meals & mixed foods** for pouches and dishes. Say what's in it by picking listed foods: logging the meal then counts as a try of each one (and of its allergens), and their age, choking and allergen advice carries over. A reaction to a meal stays with the meal, with a note on each food in it. Tick any other allergens on the label, and keep the printed ingredients and notes. With sync on, both phones see your foods. Optionally, **read a label from a photo** to fill the form in (see below).
- **What's next.** Before 6 months, a countdown and a readiness checklist you both can tick. After that, ideas for what to offer next: allergens due again, the next allergen to introduce, your to-try list and iron-rich foods.
- **Progress.** A ring for foods tried against your goal (100 before the first birthday by default), the nine common allergens as a grid (with a nudge when one hasn't been offered in a week), a to-try list, progress by category, and recent tries grouped by day, showing who logged each one.
- **Guide.** Sections that fold open: readiness signs (as a shared checklist), safe set-up, gagging vs. choking, sizes and shapes by age, common choking hazards, allergen introduction, reaction signs, foods to avoid, iron, and how much to offer.
- **Search and filters** (search stays at the top as you scroll, with a bar to jump between categories; the rest is behind the Filters button) by name (including other names like *courgette* and *capsicum*), category, allergen, what's safe at your baby's age, iron-rich, choking hazards, and tried/liked/still learning/reaction/to-try.

## About the food information

The food notes in `js/data/` were written for this app. They are general
guidance based on widely published advice from pediatric and public-health
bodies (AAP, NHS, FDA and others). The app does not copy Solid Starts' content:
it's their copyrighted work, their terms of use don't allow automated
collection of it, and this repository is public. Each food links to the
matching Solid Starts page instead. `npm run check:links` checks those links
against the sitemap Solid Starts publishes for search engines.

This is not medical advice. Talk to your pediatrician about readiness, allergy
risk (especially with severe eczema or an existing food allergy) and any concerns.

## Using it

There are three ways to run it. All use the same code.

| | Who can open it | Do both phones share one log? | Setup |
| --- | --- | --- | --- |
| GitHub Pages + Firebase sync | Anyone with the link can open the app; only your two accounts can see or change the log | Yes, live, works offline | About 15 minutes, free |
| GitHub Pages on its own | Anyone with the link | No, each phone has its own log (swap backups to merge) | About 5 minutes |
| claude.ai Artifact | Only people you share it with | Yes, live | Both of you need claude.ai accounts |

### Publish on GitHub Pages

1. In the repository, go to **Settings → Pages → Build and deployment** and set **Source** to **GitHub Actions**.
2. The workflow in `.github/workflows/pages.yml` runs the tests and publishes the site to `https://<your-username>.github.io/solid-foods-app/` on every push to the repository's default branch. You can also run it by hand from the **Actions** tab (**Test and deploy → Run workflow**).
3. Open that address on each phone and add it to the home screen. On Android, tap **Install** on the banner in the app (or **Install First Bites** on the Baby tab); on iPhone, use Safari → Share → **Add to Home Screen**. It works offline.

**About privacy:** on GitHub's Free and Pro plans, a GitHub Pages site is always public. Only GitHub Enterprise can limit who opens it. The app asks search engines not to list it, but anyone with the link can open the food guide. Your baby's log is never part of the site: it stays on each phone, or, with sync set up, in your own Firebase database, which only your two accounts can read.

### Syncing between your phones (Firebase)

Firebase is Google's app database service. The free plan is plenty for this app.

1. **Create a project.** Go to <https://console.firebase.google.com>, click **Create a project**, name it (e.g. `first-bites`), and turn off Google Analytics (not needed).
2. **Add a database.** Under **Build → Firestore Database**, click **Create database**, pick a location near you, and start in **production mode**.
3. **Lock it to the two of you.** Open the **Rules** tab, paste the contents of [`firestore.rules`](firestore.rules), replace the two example email addresses with yours and your partner's, and click **Publish**.
4. **Create your logins.** Under **Build → Authentication**, click **Get started**, enable **Email/Password**, then on the **Users** tab click **Add user** once for each of you. To stop anyone else making an account, go to **Settings → User actions** and untick **Enable create (sign-up)**.
5. **Allow your site.** In **Authentication → Settings → Authorized domains**, add `<your-username>.github.io`.
6. **Connect the app.** In **Project settings → General → Your apps**, click the web icon (`</>`), register an app (no hosting needed), and copy the `firebaseConfig` object. Paste it into [`js/sync-config.js`](js/sync-config.js) in place of `null`, for example:

   ```js
   export const FIREBASE_CONFIG = {
     apiKey: "AIza…",
     authDomain: "first-bites-1234.firebaseapp.com",
     projectId: "first-bites-1234",
     storageBucket: "first-bites-1234.appspot.com",
     messagingSenderId: "1234567890",
     appId: "1:1234567890:web:abc123",
   };
   ```

   This config isn't a secret (it only names your project), so it's fine in a public repository. The rules from step 3 are what keep the data private.
7. Commit the change to the default branch. Once the site redeploys, open the **Baby** tab on each phone and sign in. If a phone already has tries saved from before, tap **Move them to the shared log**.

After that, a try logged on either phone shows up on the other within seconds. Logging works offline too, and catches up when the phone reconnects.

2. **Get a reCAPTCHA key** (App Check uses it to recognise your site). Go to the [reCAPTCHA admin console](https://www.google.com/recaptcha/admin/create), sign in with the same Google account, and create a key: label it *First Bites*, choose **Score based (v3)**, and add the domain `<your-username>.github.io`. Copy the **site key** and the **secret key**.
3. **Register your site with App Check.** In the Firebase console, go to **Security → App Check → Apps**, find your web app, choose **reCAPTCHA**, paste the **secret key** and click **Save**.

//
// The app's own files are fetched from the network first, so a newly
// published version shows up on the next load; the cached copy is only used
// when offline. Google Fonts come from the cache once downloaded. Everything
// else (Firebase sync traffic, links out) is left alone: caching those would
// serve stale data.

const SHELL = [

const FONT_HOSTS = new Set(["fonts.googleapis.com", "fonts.gstatic.com"]);

self.addEventListener("install", (event) => {
  // cache: "reload" skips the browser's HTTP cache, so the precache holds
  // what is on the server now, not a copy from a few minutes ago.
  event.waitUntil(
    caches
      .open(CACHE)
      .then((cache) => cache.addAll(SHELL.map((url) => new Request(url, { cache: "reload" }))))
      .then(() => self.skipWaiting()),
  );
});

self.addEventListener("activate", (event) => {
  event.waitUntil(
    caches
      .keys()
      .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k))))
      .then(() => self.clients.claim()),
  );
});

self.addEventListener("fetch", (event) => {
  const { request } = event;
  if (request.method !== "GET") return;
  const url = new URL(request.url);
  if (url.origin === self.location.origin) event.respondWith(networkFirst(request));
  else if (FONT_HOSTS.has(url.hostname)) event.respondWith(cacheFirst(request));
});

async function networkFirst(request) {
  const cache = await caches.open(CACHE);
  try {
    // "no-cache" asks the server whether the file changed (cheap when it hasn't).
    const response = await fetch(request, { cache: "no-cache" });
    if (response.ok) cache.put(request, response.clone());
    return response;
  } catch (err) {
    const cached = (await cache.match(request)) || (request.mode === "navigate" && (await cache.match("./")));
    if (cached) return cached;
    throw err;
  }
}

async function cacheFirst(request) {
  const cache = await caches.open(CACHE);
  const cached = await cache.match(request);
  if (cached) return cached;
  const response = await fetch(request);
  if (response.ok || response.type === "opaque") cache.put(request, response.clone());
  return response;
}

// Shown as a popover where supported, so it sits above an open food page or
// form. It moves into the topmost open dialog, because everything outside a
// modal dialog is inert and its Undo button couldn't be tapped there.
let toastTimer;
let toastUndo = null;
const toastEl = $("#toast");
const canPopover = typeof toastEl.showPopover === "function";

function showToastIn(host) {
  if (canPopover && toastEl.matches(":popover-open")) toastEl.hidePopover();
  if (toastEl.parentElement !== host) host.append(toastEl);
  toastEl.hidden = false;
  if (canPopover) toastEl.showPopover();
}

// Replace a dialog's content without losing a message shown inside it.
function setDialogHTML(dialog, html) {
  const keep = dialog.contains(toastEl) && !toastEl.hidden;
  dialog.innerHTML = html;
  if (keep) showToastIn(dialog);
}

function toastHost() {
  const open = document.querySelectorAll("dialog[open]");
  return open.length ? open[open.length - 1] : document.body;
}

function toast(message, { undo } = {}) {
  toastUndo = undo || null;
  toastEl.innerHTML = `<span>${esc(message)}</span>${undo ? '<button type="button" class="toast-action" data-action="undo">Undo</button>' : ""}`;
  showToastIn(toastHost());
  clearTimeout(toastTimer);
  toastTimer = setTimeout(hideToast, undo ? 6000 : 3200);
}

function hideToast() {
  if (canPopover && toastEl.matches(":popover-open")) toastEl.hidePopover();
  toastEl.hidden = true;
  toastUndo = null;
}

async function undoLast() {
  const undo = toastUndo;
  hideToast();
  if (!undo) return;
  try {
    await undo();
    toast("Undone");
  } catch (err) {
    toast(describeStoreError(err));
  }
}

function statusOf(food) {
  const s = app.summaries.get(food.id);
  return s ? s.status : "new";
}

function matchesStatus(food, status) {
  const st = statusOf(food);
  switch (status) {
    case "new":
      return st === "new";
    case "tried":
      return st !== "new";
    case "queued":
      return !!app.state.queue[food.id];
    case "all":
      return true;
    default:
      return st === status;
  }
}

function matchesFilters(food, { skipStatus = false } = {}) {
  const f = app.filters;
  if (!skipStatus && !matchesStatus(food, f.status)) return false;
  if (f.cat !== "all" && food.cat !== f.cat) return false;
  if (f.allergen === "any" && !food.allergens.length) return false;
  if (f.allergen !== "all" && f.allergen !== "any" && !food.allergens.includes(f.allergen)) return false;
  if (f.flag === "ready" && isWaiting(food)) return false;
  if (f.flag === "wait" && !isWaiting(food)) return false;
  if (f.flag === "iron" && !food.iron) return false;
  if (f.flag === "hazard" && food.choking !== "high") return false;
  return L.matchesQuery(food, f.q);
}

async function toggleReadySign(input) {

// Highlight the category in view, show Back to top, and shade the sticky bars
// once the list is scrolling under them.
function updateScrollUI() {
  const toTop = $("#to-top");
  if (!toTop) return;
  toTop.hidden = scrollY < innerHeight * 1.5;
  const bar = $("#search-bar");
  const nav = $("#cat-jump");
  const barBottom = bar.getBoundingClientRect().bottom;
  const navRect = nav.getBoundingClientRect();
  const navStuck = !nav.hidden && $("#food-results").getBoundingClientRect().top < navRect.bottom - 1;
  bar.classList.toggle("is-stuck", !navStuck && $(".filters", main).getBoundingClientRect().top < barBottom - 1);
  nav.classList.toggle("is-stuck", navStuck);
  if (nav.hidden) return;
  const line = navRect.bottom + 16;
  let current = null;
  for (const group of document.querySelectorAll(".group")) {
    if (group.getBoundingClientRect().top > line) break;
    current = group.dataset.cat;
  }
  current ||= document.querySelector(".group")?.dataset.cat;
  for (const chip of nav.querySelectorAll(".chip")) {
    const on = chip.dataset.group === current;
    if (on && chip.getAttribute("aria-current") !== "true") {
      // Leave room for the edge fade, so the current chip shows clearly.
      const row = $("#cat-chips");
      row.scrollTo({ left: chip.offsetLeft - row.offsetLeft - 56, behavior: smooth() });
    }
    if (on) chip.setAttribute("aria-current", "true");
    else chip.removeAttribute("aria-current");
  }
}

// Compact tile: icon, name, and small badges that flow after the name. The
// age only shows when a food isn't ready yet.
function tileHTML(food) {
  const s = app.summaries.get(food.id);
  const status = s ? s.status : "new";
  const waiting = isWaiting(food);
  const tags = [];
  if (food.age == null) tags.push('<span class="tag tag-bad" title="Avoid for young children">Avoid</span>');
  else if (waiting) tags.push(`<span class="tag tag-wait" title="${esc(L.formatAgeRule(food))}">${ageTag(food)}</span>`);
  if (food.choking === "high") tags.push(`<span class="tag tag-icon" title="Common choking hazard: prepare carefully">${ICON.hazard}</span>`);
  if (food.iron) tags.push('<span class="tag" title="Iron-rich">Fe</span>');
  if (food.allergens.length) {
    const more = food.allergens.length > 1 ? ` +${food.allergens.length - 1}` : "";
    tags.push(`<span class="tag tag-solid" title="Common allergens: ${esc(food.allergens.map((a) => ALLERGEN_LABEL[a]).join(", "))}">${esc(ALLERGEN_SHORT[food.allergens[0]])}${more}</span>`);
  }
  if (app.state.queue[food.id]) tags.push(`<span class="tag tag-icon" title="On your to-try list">${ICON.bookmark}</span>`);
  const markLabel = s
    ? `${food.name}: tried ${plural(s.count, "time")}, ${STATUS_LABEL[status].toLowerCase()}. Log another try`
    : `Log a first try of ${food.name}`;
  const ageTitle = food.age == null ? "Not recommended for young children" : `${L.formatAgeRule(food)}${waiting ? " (not yet for your baby)" : ""}`;
  const extras = [food.choking === "high" && "Common choking hazard", food.iron && "Iron-rich", food.allergens.length && `Contains ${food.allergens.map((a) => ALLERGEN_LABEL[a].toLowerCase()).join(", ")}`].filter(Boolean);
  const mark = MARK[status] + (s && s.count > 1 ? `<span class="tile-count">${s.count}</span>` : "");
  return `<div class="tile st-${status}" data-cat="${food.cat}">
    <button type="button" class="tile-open" data-open="${food.id}" aria-label="${esc([food.name, ageTitle, ...extras, STATUS_LABEL[status]].join(". "))}.">
      ${iconHTML(food, "tile-icon")}
      <span class="tile-name">${esc(food.name)}${tags.length ? `<span class="tile-tags" aria-hidden="true">${tags.join("")}</span>` : ""}</span>
    </button>
    ${canWrite() ? `<button type="button" class="tile-check${app.popFood === food.id ? " pop" : ""}" data-action="log" data-food="${food.id}" aria-label="${esc(markLabel)}">${mark}</button>` : `<span class="tile-check" aria-hidden="true">${mark}</span>`}
  </div>`;
}

function legendHTML() {
  if (!app.summaries.size) return "";
  return `<span class="legend" aria-hidden="true">${Object.entries(STATUS_SHORT)
    .map(([status, label]) => `<span class="st-${status}">${MARK[status]}${label}</span>`)
    .join("")}</span>`;
}

function sortFoods(list) {
  if (app.filters.sort === "az") return [...list].sort((a, b) => a.name.localeCompare(b.name));
  if (app.filters.sort === "recent") {
    return [...list].sort((a, b) => {
      const da = app.summaries.get(a.id)?.lastDate || "";
      const db = app.summaries.get(b.id)?.lastDate || "";
      if (da !== db) return da < db ? 1 : -1;
      return a.name.localeCompare(b.name);
    });
  }
  return list;
}

function renderFoodResults() {
  const box = $("#food-results");
  if (!box) return;
  renderStatusChips();
  const list = sortFoods(foodList.filter((food) => matchesFilters(food)));
  const tried = list.filter((food) => app.summaries.has(food.id)).length;
  renderFilterBar();
  main.classList.toggle("is-loading", loading());
  const line = `<p class="result-line"><span class="num">${plural(list.length, "food")} · ${tried} tried</span>${loading() ? '<span class="skeleton skeleton-line" role="status"><span class="visually-hidden">Loading your tracker…</span></span>' : legendHTML()}</p>`;
  const groups = app.filters.sort === "category" ? CATEGORIES.map((c) => ({ ...c, foods: list.filter((food) => food.cat === c.id) })).filter((g) => g.foods.length) : [];
  renderCatJump(groups);
  requestAnimationFrame(updateScrollUI);
  // Searching for something that isn't listed: offer to add it.
  const q = app.filters.q.trim();
  const addOwn = q && canWrite() ? `<button type="button" class="btn btn-small${list.length ? "" : " btn-primary"}" data-action="add-food" data-name="${esc(q)}">${ICON.plus}Add “${esc(q)}” as your own food</button>` : "";
  if (!list.length) {
    box.innerHTML = `${line}<div class="empty"><strong>No foods match</strong>Try a different search, or clear the filters.<div class="button-row empty-actions">${addOwn}<button type="button" class="btn btn-small" data-action="clear-filters">Clear filters</button></div></div>`;
    return;
  }
  const after = addOwn ? `<p class="add-own">Not the one you meant? ${addOwn}</p>` : "";
  if (!groups.length) {
    box.innerHTML = `${line}<div class="grid">${list.map(tileHTML).join("")}</div>${after}`;
    return;
  }
  box.innerHTML =
    line +
    groups
      .map((g) => {
        const gTried = g.foods.filter((food) => app.summaries.has(food.id)).length;
        return `<section class="group" data-cat="${g.id}" aria-labelledby="group-${g.id}">
          <div class="group-head"><h2 id="group-${g.id}">${esc(g.label)}</h2><span class="num">${gTried}/${g.foods.length} tried</span></div>
          <div class="grid">${g.foods.map(tileHTML).join("")}</div>
        </section>`;
      })
      .join("") +
    after;
}

function factsHTML(food) {
  const facts = [];
  if (food.age == null) facts.push('<span class="fact fact-bad">Avoid for young children</span>');
  else facts.push(`<span class="fact${isWaiting(food) ? " fact-warn" : " fact-good"}">${esc(L.formatAgeRule(food))}</span>`);
  // A family food without listed ingredients has no known choking risk.
  if (food.choking) {
    const choke = CHOKING[food.choking];
    facts.push(`<span class="fact ${food.choking === "high" ? "fact-bad" : food.choking === "prep" ? "fact-warn" : ""}">${esc(choke.label)}</span>`);
  }
  if (food.iron) facts.push('<span class="fact fact-iron">Iron-rich</span>');
  for (const a of food.allergens) facts.push(`<span class="fact fact-warn">Allergen: ${esc(ALLERGEN_LABEL[a])}</span>`);
  for (const a of food.mayContain || []) facts.push(`<span class="fact">May contain: ${esc(ALLERGEN_LABEL[a].toLowerCase())}</span>`);
  return `<div class="facts">${facts.join("")}</div>`;
}

// One line near the top of a food's page: how it's going so far.
function statusLineHTML(food) {
  const s = app.summaries.get(food.id);
  if (!s) return loading() ? '<span class="skeleton skeleton-line" role="status"><span class="visually-hidden">Loading your log…</span></span>' : loadError() ? "<p>Your log didn't load.</p>" : "";

    <span>Tried ${plural(s.count, "time")}${s.mealCount ? ` (${s.mealCount} in meals)` : ""}, last on ${esc(fmtDate(s.lastDate))}</span>

function serveHTML(food) {
  if (food.age == null) return "";
  const months = babyMonths();
  const nowStage = L.stageForAge(months);
  const stage = app.stage || nowStage;
  const tabs = STAGES.map(
    (s) =>
      `<button type="button" role="tab" class="stage-btn" data-stage="${s.id}" aria-selected="${s.id === stage}" aria-controls="serve-panel">${esc(s.short)}${months != null && months >= 6 && s.id === nowStage ? '<span class="now">Now</span>' : ""}</button>`,
  ).join("");
  const st = STAGES.find((s) => s.id === stage);
  const next = STAGES[STAGES.indexOf(st) + 1];
  const tooEarly = food.age >= (next ? next.from : Infinity);
  let text = food.serve[stage];
  if (tooEarly || !text) text = `<strong>Not yet.</strong> ${esc(L.formatAgeRule(food))}.`;
  else text = esc(text);
  return `<section>
    <h3>How to serve it</h3>
    <div class="stages" role="tablist" aria-label="Age">${tabs}</div>
    <p class="serve-text" id="serve-panel" role="tabpanel">${text}</p>
  </section>`;
}

// Shrink a photo to a JPEG the model can read quickly (about 1600px on the
// long side), as base64.
async function photoToJPEG(file, longest = 1600) {
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, longest / Math.max(bitmap.width, bitmap.height));
  const canvas = document.createElement("canvas");
  canvas.width = Math.round(bitmap.width * scale);
  canvas.height = Math.round(bitmap.height * scale);
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.85));
  const bytes = new Uint8Array(await blob.arrayBuffer());
  let binary = "";
  for (let i = 0; i < bytes.length; i += 0x8000) binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  return btoa(binary);
}

async function readLabelPhoto(file) {
  const status = $("#ff-read-status", foodForm);
  const button = $("#ff-photo-btn", foodForm);
  status.innerHTML = '<p class="reading"><span class="spinner" aria-hidden="true"></span>Reading the label…</p>';
  button.classList.add("is-busy");
  try {
    const base64 = await photoToJPEG(file);
    const answer = await readLabelImage(app.cloudApi, app.labelSettings, { prompt: LABEL_PROMPT, schema: labelSchema, model: LABEL_MODEL }, base64);
    const read = parseLabel(answer, FOODS);
    if (!read.readable) throw Object.assign(new Error("No label in the photo"), { code: "unreadable" });
    if (!foodForm.open || !ff) return;
    applyLabel(read);
    const notes = [];
    if (read.spotted.length) notes.push(`Also ticked ${read.spotted.map((a) => ALLERGEN_LABEL[a].toLowerCase()).join(", ")}, from words in the ingredients.`);
    if (read.unmatched.length) notes.push(`Not in the food list, so not counted: ${read.unmatched.join(", ")}.`);
    status.innerHTML = `<div class="form-note read-ok"><p><strong>Filled in from the label.</strong> Check everything against the pack before saving.</p>${notes.map((n) => `<p>${esc(n)}</p>`).join("")}</div>`;
  } catch (err) {
    console.error(err);
    status.innerHTML = `<p class="alert">${esc(describeLabelError(err))}</p>`;
  } finally {
    button.classList.remove("is-busy");
  }
}

// Fill the form from what was read, keeping anything already typed in.
function applyLabel(read) {
  const form = $("#food-form-el", foodForm);
  if (read.name && !form.foodname.value.trim()) form.foodname.value = read.name;
  if (read.brand && !form.brand.value.trim()) form.brand.value = read.brand;
  if (read.category) form.cat.value = read.category;
  if (read.ingredientsText) form.label.value = read.ingredientsText;
  for (const id of read.ingredientIds) if (!ff.ingredients.includes(id)) ff.ingredients.push(id);
  for (const a of read.contains) ff.allergens.add(a);
  for (const a of read.mayContain) ff.mayContain.add(a);
  if (read.otherAllergens.length) {
    const extra = `The label also lists: ${read.otherAllergens.join(", ")}.`;
    if (!form.notes.value.includes(extra)) form.notes.value = [form.notes.value.trim(), extra].filter(Boolean).join("\n");
  }
  if (read.mayContain.length) $(".ff-may", form).open = true;
  $("#ff-title", foodForm).textContent = form.foodname.value.trim() || "New food";
  updateFoodFormParts();
}

// Allergens the picked ingredients bring, with the first food that brings each.
function ingredientAllergens() {
  const from = new Map();
  for (const id of ff.ingredients) for (const a of FOODS_BY_ID.get(id).allergens) if (!from.has(a)) from.set(a, FOODS_BY_ID.get(id).name);
  return from;
}

function updateFoodFormParts() {
  const form = $("#food-form-el", foodForm);
  if (!form) return;
  const cat = form.cat.value || "mixed";
  const parts = ff.ingredients.map((id) => FOODS_BY_ID.get(id));
  $("#ff-picked", form).innerHTML = parts
    .map((f) => `<button type="button" class="chip chip-on" data-action="ff-remove" data-id="${f.id}" aria-label="Remove ${esc(f.name)}">${esc(f.name)}${ICON.close}</button>`)
    .join("");
  const from = ingredientAllergens();
  $("#ff-allergens", form).innerHTML = ALLERGENS.map((a) => {
    const auto = from.get(a.id);
    return `<button type="button" class="chip" data-action="ff-allergen" data-allergen="${a.id}" aria-pressed="${!!auto || ff.allergens.has(a.id)}"${auto ? " disabled" : ""}>${esc(a.label)}${auto ? ` <span class="count">from ${esc(auto.toLowerCase())}</span>` : ""}</button>`;
  }).join("");
  $("#ff-may", form).innerHTML = ALLERGENS.map((a) => `<button type="button" class="chip" data-action="ff-may" data-allergen="${a.id}" aria-pressed="${ff.mayContain.has(a.id)}">${esc(a.label)}</button>`).join("");
  const preview = L.expandCustomFood({ id: "my-preview", name: "", cat, ingredients: ff.ingredients, allergens: [], mayContain: [], brand: "", notes: "", label: "" }, FOODS_BY_ID, CATEGORY_ICONS);
  const tile = $("#ff-tile", form);
  tile.dataset.cat = cat;
  tile.innerHTML = iconHTML(preview);
  renderIngredientResults();
}

function renderIngredientResults() {
  const box = $("#ff-ing-results", foodForm);
  const q = $("#ff-ing", foodForm).value;
  const nq = L.normalizeText(q);
  if (!nq) {
    box.innerHTML = "";
    return;
  }
  const starts = (f) => (L.normalizeText(f.name).startsWith(nq) ? 0 : 1);
  const matches = FOODS.filter((f) => !ff.ingredients.includes(f.id) && L.matchesQuery(f, q))
    .sort((a, b) => starts(a) - starts(b) || a.name.localeCompare(b.name))
    .slice(0, 8);
  box.innerHTML = matches.length
    ? matches.map((f) => `<button type="button" class="chip" data-action="ff-add" data-id="${f.id}">${ICON.plus}${esc(f.name)}</button>`).join("")
    : `<p class="hint">Nothing in the food list matches. That's fine: just leave it out.</p>`;
}

async function submitFoodForm(form) {
  // (form.name is the form's own name, so the field is called foodname.)
  const name = form.foodname.value.trim();
  if (!name) return showFoodFormError("Give the food a name.", form.foodname);
  const clash = FOODS.find((f) => L.normalizeText(f.name) === L.normalizeText(name));
  if (clash) return showFoodFormError(`${clash.name} is already in the food list. Search for it on the Foods tab.`, form.foodname);
  const existing = ff.id ? app.state.foods[ff.id] : null;
  const now = Date.now();
  // Only allergens ticked by hand are kept: the ingredients' own are worked
  // out each time, so they stay right if the ingredients change.
  const from = ingredientAllergens();
  const food = L.cleanCustomFood(
    {
      ...(existing || {}),
      id: existing ? existing.id : L.newCustomId(name),
      name,
      cat: form.cat.value,
      ingredients: ff.ingredients,
      allergens: [...ff.allergens].filter((a) => !from.has(a)),
      mayContain: [...ff.mayContain],
      brand: form.brand.value,
      label: form.label.value,
      notes: form.notes.value,
      createdAt: existing ? existing.createdAt : now,
      updatedAt: now,
    },
    KNOWN_IDS,
  );
  const submit = form.querySelector('[type="submit"]');
  submit.disabled = true;
  try {
    await app.store.saveFood(food);
    foodForm.close();
    if (existing) {
      toast("Saved changes", { undo: () => app.store.saveFood(existing) });
    } else {
      toast(`Added ${food.name}`, { undo: () => app.store.deleteFood(food.id) });
      openFood(food.id);
    }
  } catch (err) {
    showFoodFormError(describeStoreError(err));
  } finally {
    submit.disabled = false;
  }
}

// Deleting a family food takes its tries with it; Undo puts them all back.
async function deleteFood(id) {
  const food = app.state.foods[id];
  if (!food) return;
  const logs = Object.fromEntries(Object.entries(app.state.logs).filter(([, l]) => l.foodId === id));
  const queued = app.state.queue[id];
  closeFood();
  try {
    await app.store.deleteFood(id);
    const n = Object.keys(logs).length;
    toast(`Deleted ${food.name}${n ? ` and its ${plural(n, "try", "tries")}` : ""}`, {
      undo: () => app.store.importBackup({ app: "first-bites", foods: { [id]: food }, logs, queue: queued ? { [id]: queued } : {} }),
    });
  } catch (err) {
    toast(describeStoreError(err));
  }
}

// A heads-up when logging a food that isn't recommended yet at this age.
function logWarningHTML(food) {
  if (food.age != null && !isWaiting(food)) return "";
  const reason = food.wait ? ` ${esc(food.wait)}` : "";
  if (food.age == null) return `<div class="wait-box is-avoid"><strong>${esc(food.name)} is best avoided for young children.</strong>${reason}</div>`;
  const months = babyMonths();
  const age = months != null ? ` ${esc(babyName() || "Your baby")} is ${esc(L.formatAgeLong(months))}.` : "";
  return `<div class="wait-box"><strong>${esc(food.name)} isn't recommended until ${esc(L.formatAgeRule(food).replace("From ", ""))}.</strong>${age}${reason}</div>`;
}

function servedParts(text) {
  return String(text || "")
    .split(",")
    .map((part) => part.trim().toLowerCase())
    .filter(Boolean);
}

// A served quick pick adds its words to the text box, or takes them out.
function toggleServed(button) {
  const input = $("#log-served", logDialog);
  const parts = String(input.value).split(",").map((part) => part.trim()).filter(Boolean);
  const at = parts.findIndex((part) => part.toLowerCase() === button.dataset.text);
  if (at >= 0) parts.splice(at, 1);
  else parts.push(button.dataset.text);
  input.value = parts.join(", ");
  updateServedPicks();
}

function updateServedPicks() {
  const served = servedParts($("#log-served", logDialog).value);
  for (const b of logDialog.querySelectorAll('[data-action="served-pick"]')) b.setAttribute("aria-pressed", String(served.includes(b.dataset.text)));
}

async function submitLog(form) {
  const data = new FormData(form);

  if (date > L.todayISO()) return showLogError("That date is in the future.", dateInput);
  if (!outcome) return showLogError("Choose how it went.", $('[name="outcome"]', form));
  const hasReaction = $("#log-has-reaction", form).checked;

    updatedAt: now,

const dayFmt = new Intl.DateTimeFormat(undefined, { weekday: "short", day: "numeric", month: "short" });

function dayLabel(iso) {
  const n = L.daysBetween(iso, L.todayISO());
  if (n === 0) return "Today";
  if (n === 1) return "Yesterday";
  const [y, m, d] = iso.split("-").map(Number);
  return dayFmt.format(new Date(y, m - 1, d));
}

function ringHTML(value, max) {
  const r = 52;
  const c = 2 * Math.PI * r;
  const filled = Math.min(1, value / max) * c;
  return `<svg class="ring" viewBox="0 0 120 120" aria-hidden="true"><circle class="ring-track" cx="60" cy="60" r="${r}"/><circle class="ring-fill" cx="60" cy="60" r="${r}" stroke-dasharray="${filled.toFixed(1)} ${c.toFixed(1)}" transform="rotate(-90 60 60)"/></svg>`;
}

// Bring every readiness checkbox and count up to date (e.g. after the other
// phone ticks one), without redrawing what someone may be tapping.

function refreshSetup() {
  const box = $("#sync-box");
  // Don't wipe the sign-in form while someone is typing in it.
  const typing = box && box.contains(document.activeElement) && $("#signin-form", box) && app.store.mode === "local";
  if (box && !typing) box.innerHTML = syncHTML();
  for (const input of document.querySelectorAll("[data-profile]")) {
    if (document.activeElement === input) continue;
    const key = input.dataset.profile;
    const value = app.state.profile[key];
    input.value = value ?? (key === "goal" ? DEFAULT_GOAL : "");
  }
}

async function saveProfileField(input) {
  const key = input.dataset.profile;
  let value = input.value.trim();
  if (key === "goal") {
    value = parseInt(value, 10);
    if (!Number.isInteger(value) || value < 1 || value > 1000) {
      input.value = goal();
      return;
    }
  }
  if (key === "birthDate" && value && (value > L.todayISO() || !/^\d{4}-\d{2}-\d{2}$/.test(value))) {
    toast("Pick a birth date in the past.");
    return;
  }
  if (app.state.profile[key] === value) return;
  try {
    await app.store.setProfile({ [key]: value });
    toast("Saved");
  } catch (err) {
    toast(describeStoreError(err));
  }
}

async function exportBackup() {
  const backup = L.makeBackup(app.state);
  const text = JSON.stringify(backup, null, 2);
  const filename = `first-bites-backup-${L.todayISO()}.json`;
  const claude = window.claude;
  if (claude && typeof claude.use === "function") {
    const downloads = await claude.use("downloads").catch(() => null);
    if (downloads) {
      try {
        await downloads.save({ filename, data: text });
        toast("Backup saved");
        return;
      } catch (err) {
        if (err && err.code === "declined") return;
      }
    }
    showExportText(text);
    return;
  }
  try {
    const url = URL.createObjectURL(new Blob([text], { type: "application/json" }));
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    document.body.append(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
    toast("Backup downloaded");
  } catch {
    showExportText(text);
  }
}

async function showExportText(text) {
  const box = $("#export-box");
  const area = $("#export-text");
  if (!box || !area) return;
  area.value = text;
  box.hidden = false;
  area.focus();
  area.select();
  try {
    await navigator.clipboard.writeText(text);
    toast("Backup copied to the clipboard");
  } catch {
    toast("Copy the backup text below");
  }
}

function showBackupError(message) {
  const el = $("#backup-error");
  if (!el) return toast(message);
  el.textContent = message;
  el.hidden = !message;
}

// Returns true when the backup was read and merged.
async function importText(text) {
  showBackupError("");
  if (!text.trim()) {
    showBackupError("Paste the backup text first.");
    return false;
  }
  let backup;
  try {
    backup = JSON.parse(text);
  } catch {
    showBackupError("That doesn't look like a First Bites backup. Choose the .json file you exported, or paste its full text.");
    return false;
  }
  try {
    const { added, updated } = await app.store.importBackup(backup);
    toast(added || updated ? `Imported ${plural(added, "new try", "new tries")}${updated ? `, updated ${updated}` : ""}` : "Already up to date: nothing new in that backup");
    return true;
  } catch (err) {
    showBackupError(err && err.code ? describeStoreError(err) : err.message);
    return false;
  }
}

function importFile(file) {
  showBackupError("");
  if (file.size > 5_000_000) return showBackupError("That file is too big to be a First Bites backup.");
  file.text().then(importText, () => showBackupError("Couldn't read that file. Try exporting the backup again."));
}

function toggleClearBox(open) {
  const box = $("#clear-box");
  if (!box) return;
  box.hidden = !open;
  $('[data-action="clear-all"]').setAttribute("aria-expanded", String(open));
  $("#clear-confirm").value = "";
  $('[data-action="clear-all-confirm"]').disabled = true;
  if (!open) return;
  const tries = Object.keys(app.state.logs).length;
  const queued = Object.keys(app.state.queue).length;
  const own = Object.keys(app.state.foods || {}).length;
  const what = [plural(tries, "logged try", "logged tries"), queued ? `${plural(queued, "food")} on the to-try list` : "", own ? plural(own, "food you added", "foods you added") : "", "your baby's details"].filter(Boolean);
  const where = app.store.mode === "local" ? "from this phone" : "from the shared log, so they'll disappear from your partner's phone too";
  $("#clear-what").innerHTML = `<strong>This can't be undone.</strong> It deletes ${esc(what.slice(0, -1).join(", "))} and ${esc(what[what.length - 1])} ${esc(where)}.`;
  $("#clear-confirm").focus();
}

async function clearAll() {

  }
}

function renderView() {
  if (app.view === "foods") renderFoods();
  else if (app.view === "progress") renderProgress();
  else if (app.view === "guide") renderGuide();
  else renderSetup();
}

function route() {
  const hash = decodeURIComponent(location.hash.slice(1));
  app.pendingFood = null;
  if (hash.startsWith("food-")) {
    const id = hash.slice(5);
    if (foodById(id)) {
      if (!main.hasChildNodes()) {
        renderChrome();
        renderView();
      }
      showSheet(id);
      return;
    }
    // One of the family's foods that hasn't arrived yet (the log is still
    // loading, or it was just added): show it once it does.
    if (L.isCustomId(id)) app.pendingFood = id;
  }
  hideSheet();
  const view = VIEWS.includes(hash) ? hash : "foods";
  if (view !== app.view || !main.hasChildNodes()) {
    app.view = view;
    renderChrome();
    renderView();
    window.scrollTo(0, 0);
  }
}

function refresh() {
  renderChrome();
  if (app.view === "foods") {
    const notices = $("#status-notices");
    if (notices) notices.innerHTML = statusNoticesHTML();
    // The how-to line is only needed until the first try is logged.
    const intro = $("#foods-intro");
    if (intro) intro.hidden = app.summaries.size > 0;
    const addFood = $("#add-food-btn");
    if (addFood) addFood.hidden = !canWrite();
    const next = $("#next-card");
    // Don't redraw a checkbox someone is tapping.
    if (next && !next.contains(document.activeElement)) next.innerHTML = nextCardHTML();
    else if (next) syncReadySigns();
    renderFoodResults();
    if (filterSheet.open) updateFilterDone();
  } else if (app.view === "progress") renderProgress();
  else if (app.view === "guide") syncReadySigns();
  else if (app.view === "setup") refreshSetup();
  if (app.openFoodId && sheet.open) {
    const food = foodById(app.openFoodId);
    // A food deleted on the other phone: close its page.
    if (!food) closeFood();
    // Redraw a family food's whole page only when the food itself changed.
    else if (food.custom && food.source !== app.sheetSource) renderSheet();
    else refreshSheetLog(food);
  }
  // A page for one of the family's foods, opened before the log loaded.
  if (app.pendingFood && foodById(app.pendingFood)) route();
}

// Resolve the people named on shared log entries.
async function resolveNames() {
  if (!app.store || app.store.mode !== "shared") return;
  const ids = [...new Set(Object.values(app.state.logs).map((l) => l.by).filter(Boolean))];
  const missing = ids.filter((id) => !(id in app.names));
  if (!missing.length) return;
  const profiles = await app.store.profiles(ids);
  let changed = false;
  for (const id of ids) {
    const name = (profiles[id] && profiles[id].name) || "";
    if (app.names[id] !== name) {
      app.names[id] = name;
      changed = true;
    }
  }
  if (changed) refresh();
}

// Drag a sheet down to close it, like a native bottom sheet. The drag starts
// from the header, or anywhere once the sheet is scrolled to the top.
function enableSwipeClose(dialog, { allow = () => true, close }) {
  let drag = null;
  const reset = () => {
    dialog.style.transition = "";
    dialog.style.transform = "";
  };
  dialog.addEventListener(
    "touchstart",
    (e) => {
      drag = null;
      if (e.touches.length !== 1 || matchMedia("(min-width: 760px)").matches) return;
      if (e.target.closest(".chip-row, .choice-row, .stages, input, textarea, select")) return;
      const fromHead = !!e.target.closest(".sheet-head");
      if (!fromHead && dialog.scrollTop > 0) return;
      drag = { y: e.touches[0].clientY, t: Date.now(), dy: 0, fromHead };
    },
    { passive: true },
  );
  dialog.addEventListener(
    "touchmove",
    (e) => {
      if (!drag) return;
      const dy = e.touches[0].clientY - drag.y;
      if (dy < 0 || (!drag.fromHead && dialog.scrollTop > 0)) {
        drag = null;
        reset();
        return;
      }
      // Dragging the header moves the sheet, not the content behind it.
      if (drag.fromHead && e.cancelable) e.preventDefault();
      drag.dy = dy;
      dialog.style.transition = "none";
      dialog.style.transform = `translateY(${dy}px)`;
    },
    { passive: false },
  );
  const end = () => {
    if (!drag) return;
    const { dy, t } = drag;
    drag = null;
    const flick = dy > 40 && dy / Math.max(1, Date.now() - t) > 0.6;
    dialog.style.transition = "transform 180ms ease-out";
    if ((dy > 110 || flick) && allow()) {
      dialog.style.transform = "translateY(100%)";
      setTimeout(close, 170);
    } else {
      dialog.style.transform = "";
    }
  };
  dialog.addEventListener("touchend", end);
  dialog.addEventListener("touchcancel", end);
  dialog.addEventListener("close", () => {
    reset();
    // Keep a visible message (and its Undo) on screen when its dialog closes.
    if (toastEl.parentElement === dialog && !toastEl.hidden) showToastIn(toastHost());
  });
}

function installDismissed() {
  try {
    return localStorage.getItem(INSTALL_DISMISSED_KEY) === "1";

    return false;

// Steps for the browser in use, for when it doesn't offer its own prompt.
function installSteps() {
  const ua = navigator.userAgent;
  const android = /android/i.test(ua);
  if (/iphone|ipad|ipod/i.test(ua) || (navigator.platform === "MacIntel" && navigator.maxTouchPoints > 1)) {
    return "Tap the Share button (the square with an arrow), then <strong>Add to Home Screen</strong>.";
  }
  if (/firefox|fxios/i.test(ua)) {
    return android
      ? "Tap the menu button (⋮), then <strong>Add app to Home screen</strong> or <strong>Add to Home screen</strong>."
      : "Firefox on a computer can't install apps like this. Bookmark this page, or open it in Chrome or Edge to install it.";
  }
  if (/samsungbrowser/i.test(ua)) return "Tap the menu button (☰), then <strong>Add page to</strong> and <strong>Home screen</strong>.";
  if (android) {
    return "Tap the menu button (⋮), then <strong>Add to home screen</strong> or <strong>Install app</strong>. If neither is there, open this page in Firefox and add it from its menu.";
  }
  return "Click the install icon at the right of the address bar, or open the browser menu and choose <strong>Install First Bites</strong>.";
}

function registerServiceWorker() {
  if (window.claude || !("serviceWorker" in navigator)) return;
  if (location.protocol !== "https:" && location.hostname !== "localhost") return;
  // When a newer version of the app takes over, reload once so this page
  // isn't left running the old files.
  const hadController = !!navigator.serviceWorker.controller;
  let reloading = false;
  navigator.serviceWorker.addEventListener("controllerchange", () => {
    if (!hadController || reloading) return;
    reloading = true;
    location.reload();
  });
  navigator.serviceWorker.register("sw.js").catch(() => {});
}

// Switch the app to a different store (e.g. after signing in to sync).
function attachStore(store) {
  if (app.unsubscribe) app.unsubscribe();
  if (app.store && app.store !== store && app.store.dispose) app.store.dispose();
  app.store = store;
  store.onError = (err) => toast(describeStoreError(err));
  if (store.start && store.mode === "cloud") store.start();
  app.unsubscribe = store.subscribe((state) => {
    // Same data with a new sync status (e.g. a write reached the server):
    // only the Baby tab's sync note needs redrawing.
    if (state === app.state && !store.error) {
      if (app.view === "setup") refreshSetup();
      return;
    }
    app.state = state;
    rebuildFoods();
    app.summaries = L.summarizeAll(state);
    refresh();
    resolveNames();
  });
  if (app.view === "setup") renderSetup();
}

// Firebase sync for the GitHub Pages version (see js/sync-config.js).
async function startCloud() {
  app.cloud = { status: "loading" };
  if (app.view === "setup") refreshSetup();
  try {
    const cloud = await loadCloud(app.syncConfig);
    app.cloudApi = cloud;
    cloud.onUser((user) => {
      if (user) {
        if (!(app.store.mode === "cloud" && app.store.user.uid === user.uid)) attachStore(new CloudStore(cloud, user, KNOWN_IDS));
        app.cloud = { status: "signed-in" };
      } else {
        if (app.store.mode === "cloud") attachStore(new LocalStore(KNOWN_IDS));
        app.cloud = { status: "signed-out" };
      }
      refresh();
    });
  } catch (err) {
    app.cloud = { status: "error", message: navigator.onLine ? "The sync service couldn't be loaded." : "You're offline." };
    refresh();
  }
}

async function signIn(form) {
  const email = form.email.value.trim();
  const password = form.password.value;
  const error = $("#signin-error");
  const submit = form.querySelector('[type="submit"]');
  submit.disabled = true;
  error.hidden = true;
  try {
    await app.cloudApi.signIn(email, password);
    toast("Signed in. Your log now syncs.");
  } catch (err) {
    error.textContent = describeAuthError(err);
    error.hidden = false;
  } finally {
    submit.disabled = false;
  }
}

  const error = $("#signin-error");
  if (!email) {
    error.textContent = "Enter your email above, then tap Forgot password again.";
    error.hidden = false;

    await app.cloudApi.resetPassword(email);
    toast("Check your email for a link to reset your password.");
  } catch (err) {
    error.textContent = describeAuthError(err);
    error.hidden = false;
  }
}

// Move tries logged while signed out into the shared log. They leave this
// phone's own log, so they're only offered once.
async function copyLocalToCloud() {
  if (!unsyncedLocalIds().length) return;
  const local = readLocalState(KNOWN_IDS);
  try {
    const { added } = await app.store.importBackup(L.makeBackup(local));
    removeLocalLogs(Object.keys(local.logs), KNOWN_IDS);
    toast(`Moved ${plural(added, "try", "tries")} to the shared log`);
    refresh();
  } catch (err) {
    toast(describeStoreError(err));
  }
}

async function boot() {
  mountIconSprite();
  route();
  const store = await openStore(KNOWN_IDS);
  // Read the sync settings before showing this phone's log, so the page
  // knows synced details are on their way and doesn't flash prompts.
  if (store.mode === "local" && !window.claude) {
    const { config, error, labels } = await readSyncConfig();
    app.syncConfig = config;
    app.labelSettings = labels || null;
    app.syncConfigError = error || "";
    if (error) console.error(error);
    if (config) app.cloud = { status: "loading" };
  }
  attachStore(store);
  if (app.syncConfig) startCloud();
  registerServiceWorker();
}
